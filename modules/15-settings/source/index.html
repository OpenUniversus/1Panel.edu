 452func checkEntrancePattern(val string) bool {
 453	if len(val) == 0 {
 454		return true
 455	}
 456	result, _ := regexp.MatchString("^[a-zA-Z0-9]{5,116}$", val)
 457	if !result {
 458		return false
 459	}
 460	lowerVal := strings.ToLower(val)
 461	for key := range constant.WebUrlMap {
 462		if key == "/" || !strings.HasPrefix(key, "/") {
 463			continue
 464		}
 465		if strings.Count(key, "/") != 1 {
 466			continue
 467		}
 468		segment := strings.ToLower(strings.TrimPrefix(key, "/"))
 469		if len(segment) < 5 {
 470			continue
 471		}
 472		if lowerVal == segment {
 473			return false
 474		}
 475	}
 476	assetsList := [2]string{"public", "assets"}
 477	for _, item := range assetsList {
 478		if lowerVal == item {
 479			return false
 480		}
 481	}
 482	return true
 483}

 485func normalizePasskeyTrustedProxies(value string) (string, error) {
 486	if strings.TrimSpace(value) == "" {
 487		return "", nil
 488	}
 489	lines := strings.Split(value, "\n")
 490	validLines := make([]string, 0, len(lines))
 491	for _, line := range lines {
 492		line = strings.TrimSpace(line)
 493		if line == "" {
 494			continue
 495		}
 496		validLines = append(validLines, line)
 497	}
 498	if len(validLines) == 0 {
 499		return "", nil
 500	}
 501	normalized := strings.Join(validLines, "\n")
 502	if _, err := common.HandleIPList(normalized); err != nil {
 503		return "", err
 504	}
 505	return normalized, nil

 780func checkProxy(req dto.ProxyUpdate) error {
 781	var transport http.Transport
 782	proxyItem := fmt.Sprintf("%s:%s", req.ProxyUrl, req.ProxyPort)
 783	switch req.ProxyType {
 784	case "http", "https":
 785		proxyURL, err := url.Parse(proxyItem)
 786		if err != nil {
 787			return buserr.WithErr("ErrProxySetting", fmt.Errorf("parse url %s failed, err: %v", proxyItem, err))
 788		}
 789		if len(req.ProxyUser) != 0 {
 790			proxyURL.User = url.UserPassword(req.ProxyUser, req.ProxyPasswd)
 791		}
 792		transport = http.Transport{Proxy: http.ProxyURL(proxyURL), TLSClientConfig: &tls.Config{InsecureSkipVerify: true}}
 793	case "socks5":
 794		var auth *proxy.Auth
 795		if len(req.ProxyUser) == 0 {
 796			auth = nil
 797		} else {
 798			auth = &proxy.Auth{User: req.ProxyUser, Password: req.ProxyPasswd}
 799		}
 800		dialer, err := proxy.SOCKS5("tcp", proxyItem, auth, proxy.Direct)
 801		if err != nil {
 802			return buserr.WithErr("ErrProxySetting", fmt.Errorf("new socks5 proxy failed, err: %v", err))
 803		}
 804		dialContext := func(ctx context.Context, network, addr string) (net.Conn, error) {
 805			return dialer.Dial(network, addr)
 806		}
 807		transport = http.Transport{DialContext: dialContext}
 808	case "", "close":
 809		return nil
 810	default:
 811		return buserr.WithName("ErrNotSupportType", req.ProxyType)
 812	}
 813	defer func() {
 814		if r := recover(); r != nil {
 815			global.LOG.Errorf("handle request failed, error message: %v", r)
 816			return
 817		}
 818	}()
 819
 820	client := http.Client{Timeout: 3 * time.Second, Transport: &transport}
 821	ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
 822	defer cancel()
 823	request, err := http.NewRequestWithContext(ctx, http.MethodGet, "https:">//1panel.cn/", nil)
 824	if err != nil {
 825		return buserr.WithErr("ErrProxySetting", err)
 826	}
 827	request.Header.Set("Content-Type", "application/json")
 828	resp, err := client.Do(request)
 829	if err != nil {
 830		return buserr.WithErr("ErrProxySetting", err)
 831	}
 832	defer resp.Body.Close()
 833	if _, err := io.ReadAll(resp.Body); err != nil {
 834		return buserr.WithErr("ErrProxySetting", err)
 835	}
 836	return nil
 837}
