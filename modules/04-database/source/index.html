 326		}

 105func (u *MongodbService) LoadFromRemote(req dto.MongodbLoadDB) error {
 106	databases, err := mongodbRepo.List(mongodbRepo.WithByMongodbName(req.Database))
 107	if err != nil {
 108		return err
 109	}
 110
 111	datas, err := loadMongodbDatabases(req)
 112	if err != nil {
 113		return err
 114	}
 115	global.LOG.Infof("sync mongodb databases from %s:%s, found %d items", req.From, req.Database, len(datas))
 116
 117	deleteList := append([]model.DatabaseMongodb(nil), databases...)
 118	for _, data := range datas {
 119		hasOld := false
 120		for i := 0; i < len(databases); i++ {
 121			if strings.EqualFold(databases[i].Name, data.Name) && strings.EqualFold(databases[i].MongodbName, req.Database) {
 122				hasOld = true
 123				updateMap := map[string]interface{}{"is_delete": false}
 124				if len(data.Username) != 0 {
 125					updateMap["username"] = data.Username
 126				}
 127				_ = mongodbRepo.Update(databases[i].ID, updateMap)
 128				for j := 0; j < len(deleteList); j++ {
 129					if deleteList[j].ID == databases[i].ID {
 130						deleteList = append(deleteList[:j], deleteList[j+1:]...)
 131						break
 132					}
 133				}
 134				break
 135			}
 136		}
 137		if !hasOld {
 138			createItem := model.DatabaseMongodb{
 139				Name:        data.Name,
 140				From:        req.From,
 141				MongodbName: req.Database,
 142				Username:    data.Username,
 143				Password:    "",
 144				Description: "",
 145			}
 146			if err := mongodbRepo.Create(context.Background(), &createItem); err != nil {
 147				return err
 148			}
 149		}
 150	}
 151	for _, delItem := range deleteList {
 152		_ = mongodbRepo.Update(delItem.ID, map[string]interface{}{"is_delete": true})
 153	}
 154	return nil

 931		beginMark = "__1panel_json_begin__"
 932		endMark   = "__1panel_json_end__"
 933	)
 934
 935	output := strings.TrimSpace(stdout)
 936	if output == "" {
 937		return nil, fmt.Errorf("empty mongodb command output")
 938	}
 939
 940	beginIndex := strings.Index(output, beginMark)
 941	endIndex := strings.LastIndex(output, endMark)
 942	if beginIndex >= 0 && endIndex > beginIndex {
 943		content := strings.TrimSpace(output[beginIndex+len(beginMark) : endIndex])
 944		if content != "" {
 945			return []byte(content), nil
 946		}
 947	}
 948
 949	lines := strings.Split(output, "\n")
 950	for i := len(lines) - 1; i >= 0; i-- {
 951		line := strings.TrimSpace(lines[i])
 952		if line == "" {
 953			continue
 954		}
 955		return []byte(line), nil
 956	}
 957	return nil, fmt.Errorf("mongodb command output does not contain json result")

 627func (u *MysqlService) CreateUser(req dto.MysqlUserCreate) error {
 628	if cmd.CheckIllegal(req.Username, req.Password, req.Host) {
 629		return buserr.New("ErrCmdIllegal")
 630	}
 631	dbs := make([]string, 0, len(req.DBs))
 632	dbSet := make(map[string]struct{}, len(req.DBs))
 633	for _, db := range req.DBs {
 634		if db == "" {
 635			continue
 636		}
 637		if cmd.CheckIllegal(db) {
 638			return buserr.New("ErrCmdIllegal")
 639		}
 640		if db == "*" {
 641			return errors.New("global mysql privileges must be managed outside 1Panel")
 642		}
 643		if _, ok := dbSet[db]; ok {
 644			continue
 645		}
 646		dbSet[db] = struct{}{}
 647		dbs = append(dbs, db)
 648	}
 649	if err := checkMysqlNormalUser(req.Username); err != nil {
 650		return err
 651	}
 652	hosts, err := parseMysqlHosts(req.Host, true)
 653	if err != nil {
 654		return err
 655	}
 656	dbType, err := resolveDatabaseUserType(req.Database)
 657	if err != nil {
 658		return err
 659	}
 660	cli, version, err := LoadMysqlClientByFrom(req.Database)
 661	if err != nil {
 662		return err
 663	}
 664	defer cli.Close()
 665	createdHosts := make([]string, 0, len(hosts))
 666	rollbackCreatedHosts := func() {
 667		for i := len(createdHosts) - 1; i >= 0; i-- {
 668			host := createdHosts[i]
 669			if rollbackErr := cli.DeleteUser(client.UserInfo{Username: req.Username, Host: host}, version, 300); rollbackErr != nil {
 670				global.LOG.Errorf("rollback mysql user %s@%s failed, err: %v", req.Username, host, rollbackErr)
 671			}
 672		}
 673	}
 674	savedHosts := make([]string, 0, len(hosts))
 675	rollbackSavedHosts := func() {
 676		for i := len(savedHosts) - 1; i >= 0; i-- {
 677			host := savedHosts[i]
 678			if rollbackErr := databaseUserRepo.Delete(
 679				repo.WithByType(dbType),
 680				databaseUserRepo.WithByDatabase(req.Database),
 681				databaseUserRepo.WithByUser(req.Username, host),
 682			); rollbackErr != nil {
 683				global.LOG.Errorf("rollback mysql user record %s@%s failed, err: %v", req.Username, host, rollbackErr)
 684			}
 685		}
 686	}
 687	grantedItems := make([]client.GrantInfo, 0, len(dbs)*len(hosts))
 688	rollbackGrantedItems := func() {
 689		for i := len(grantedItems) - 1; i >= 0; i-- {
 690			item := grantedItems[i]
 691			if rollbackErr := cli.RevokeGrant(item, 300); rollbackErr != nil {
 692				global.LOG.Errorf(
 693					"rollback mysql grant %s to %s@%s failed, err: %v",
 694					item.Database,
 695					item.Username,
 696					item.Host,
 697					rollbackErr,
 698				)
 699			}
 700			if rollbackErr := databaseUserGrantRepo.Delete(
 701				repo.WithByType(dbType),
 702				databaseUserGrantRepo.WithByDatabase(req.Database),
 703				databaseUserGrantRepo.WithByDBName(item.Database),
 704				databaseUserGrantRepo.WithByUser(item.Username, item.Host),
 705			); rollbackErr != nil {
 706				global.LOG.Errorf(
 707					"rollback mysql grant record %s to %s@%s failed, err: %v",
 708					item.Database,
 709					item.Username,
 710					item.Host,
 711					rollbackErr,
 712				)
 713			}
 714		}
 715	}
 716	rollbackAll := func() {
 717		rollbackGrantedItems()
 718		rollbackSavedHosts()
 719		rollbackCreatedHosts()
 720	}
 721	for _, host := range hosts {
 722		if err := cli.CreateUserOnly(client.UserInfo{Username: req.Username, Host: host}, req.Password, 300); err != nil {
 723			rollbackCreatedHosts()
 724			return err
 725		}
 726		createdHosts = append(createdHosts, host)
 727	}
 728	for _, host := range hosts {
 729		if err := saveDatabaseUserCredential(dbType, req.Database, req.Username, host, req.Password, req.Description); err != nil {
 730			rollbackAll()
 731			return err
 732		}
 733		savedHosts = append(savedHosts, host)
 734	}
 735	for _, db := range dbs {
 736		for _, host := range hosts {
 737			item := client.GrantInfo{Database: db, Username: req.Username, Host: host}
 738			if err := cli.GrantUser(item, 300); err != nil {
 739				rollbackAll()
 740				return err
 741			}
 742			grantedItems = append(grantedItems, item)
 743			if err := saveDatabaseUserGrant(dbType, req.Database, db, req.Username, host); err != nil {
 744				rollbackAll()
 745				return err
 746			}
 747		}
 748	}
 749	return nil
 750}

1502func updateMyCnf(oldFiles []string, group string, param string, value interface{}) []string {
1503	isOn := false
1504	hasGroup := false
1505	hasKey := false
1506	regItem := re.GetRegex(re.MysqlGroupPattern)
1507	var newFiles []string
1508	i := 0
1509	for _, line := range oldFiles {
1510		i++
1511		if strings.HasPrefix(line, group) {
1512			isOn = true
1513			hasGroup = true
1514			newFiles = append(newFiles, line)
1515			continue
1516		}
1517		if !isOn {
1518			newFiles = append(newFiles, line)
1519			continue
1520		}
1521		if strings.HasPrefix(line, param+"=") || strings.HasPrefix(line, "# "+param+"=") {
1522			newFiles = append(newFiles, fmt.Sprintf("%s=%v", param, value))
1523			hasKey = true
1524			continue
1525		}
1526		if regItem.Match([]byte(line)) || i == len(oldFiles) {
1527			isOn = false
1528			if !hasKey {
1529				newFiles = append(newFiles, fmt.Sprintf("%s=%v", param, value))
1530			}
1531			newFiles = append(newFiles, line)
1532			continue
1533		}
1534		newFiles = append(newFiles, line)
1535	}
1536	if !hasGroup {
1537		newFiles = append(newFiles, group+"\n")
1538		newFiles = append(newFiles, fmt.Sprintf("%s=%v\n", param, value))
1539	}
1540	return newFiles
1541}

 227func confSet(redisName string, redisType string, updateType string, changeConf []redisConfig) error {
 228	path := fmt.Sprintf("%s/%s/%s/conf/redis.conf", global.Dir.AppInstallDir, redisType, redisName)
 229	lineBytes, err := os.ReadFile(path)
 230	if err != nil {
 231		return err
 232	}
 233	files := strings.Split(string(lineBytes), "\n")
 234
 235	startIndex, endIndex, emptyLine := 0, 0, 0
 236	var newFiles []string
 237	for i := 0; i < len(files); i++ {
 238		if files[i] == "# Redis configuration rewrite by 1Panel" {
 239			startIndex = i
 240			newFiles = append(newFiles, files[i])
 241			continue
 242		}
 243		if files[i] == "# End Redis configuration rewrite by 1Panel" {
 244			endIndex = i
 245			break
 246		}
 247		if startIndex == 0 && strings.HasPrefix(files[i], "save ") {
 248			newFiles = append(newFiles, "#   "+files[i])
 249			continue
 250		}
 251		if startIndex != 0 && endIndex == 0 && (len(files[i]) == 0 || (updateType == "rbd" && strings.HasPrefix(files[i], "save "))) {
 252			emptyLine++
 253			continue
 254		}
 255		newFiles = append(newFiles, files[i])
 256	}
 257	if startIndex == 0 {
 258		newFiles = append(newFiles, "# Redis configuration rewrite by 1Panel")
 259		startIndex = len(newFiles) - 1
 260	}
 261	endIndex = endIndex - emptyLine
 262	for _, item := range changeConf {
 263		if item.key == "save" {
 264			saveVal := strings.Split(item.value, ",")
 265			for i := 0; i < len(saveVal); i++ {
 266				newFiles = append(newFiles, "save "+saveVal[i])
 267			}
 268			continue
 269		}
 270
 271		isExist := false
 272		for i := startIndex; i < endIndex; i++ {
 273			if strings.HasPrefix(newFiles[i], item.key) || strings.HasPrefix(newFiles[i], "# "+item.key) {
 274				if item.value == "0" || len(item.value) == 0 {
 275					newFiles[i] = fmt.Sprintf("# %s %s", item.key, item.value)
 276				} else {
 277					newFiles[i] = fmt.Sprintf("%s %s", item.key, item.value)
 278				}
 279				isExist = true
 280				break
 281			}
 282		}
 283		if !isExist {
 284			if item.value == "0" || len(item.value) == 0 {
 285				newFiles = append(newFiles, fmt.Sprintf("# %s %s", item.key, item.value))
 286			} else {
 287				newFiles = append(newFiles, fmt.Sprintf("%s %s", item.key, item.value))
 288			}
 289		}
 290	}
 291	newFiles = append(newFiles, "# End Redis configuration rewrite by 1Panel")
 292
 293	file, err := os.OpenFile(path, os.O_WRONLY|os.O_TRUNC, constant.FilePerm)
 294	if err != nil {
 295		return err
 296	}
 297	defer file.Close()
 298	_, err = file.WriteString(strings.Join(newFiles, "\n"))
 299	if err != nil {
 300		return err
 301	}
 302	return nil
 303}
