 859		}
 860		if !fileOp.Stat(req.CertificatePath) {
 861			return buserr.New("ErrSSLCertificateNotFound")
 862		}
 863		if content, err := fileOp.GetContent(req.PrivateKeyPath); err != nil {
 864			return err
 865		} else {
 866			websiteSSL.PrivateKey = string(content)
 867		}
 868		if content, err := fileOp.GetContent(req.CertificatePath); err != nil {
 869			return err
 870		} else {
 871			websiteSSL.Pem = string(content)
 872		}
 873		websiteSSL.CertPath = req.CertificatePath
 874		websiteSSL.PrivateKeyPath = req.PrivateKeyPath
 875	} else {
 876		websiteSSL.PrivateKey = req.PrivateKey
 877		websiteSSL.Pem = req.Certificate
 878		websiteSSL.CertPath = ""
 879		websiteSSL.PrivateKeyPath = ""
 880	}
 881
 882	privateKeyCertBlock, _ := pem.Decode([]byte(websiteSSL.PrivateKey))
 883	if privateKeyCertBlock == nil {
 884		return buserr.New("ErrSSLKeyFormat")
 885	}
 886
 887	var (
 888		cert    *x509.Certificate
 889		pemData = []byte(websiteSSL.Pem)
 890	)
 891	for {
 892		certBlock, reset := pem.Decode(pemData)
 893		if certBlock == nil {
 894			break
 895		}
 896		cert, err = x509.ParseCertificate(certBlock.Bytes)
 897		if err != nil {
 898			return err
 899		}
 900		if len(cert.DNSNames) > 0 || len(cert.IPAddresses) > 0 {
 901			break
 902		}
 903		pemData = reset
 904	}
 905	if pemData == nil || cert == nil {
 906		return buserr.New("ErrSSLCertificateFormat")
 907	}
 908
 909	websiteSSL.ExpireDate = cert.NotAfter
 910	websiteSSL.StartDate = cert.NotBefore
 911	websiteSSL.Type = cert.Issuer.CommonName
 912	if len(cert.Issuer.Organization) > 0 {
 913		websiteSSL.Organization = cert.Issuer.Organization[0]
 914	} else {
 915		websiteSSL.Organization = cert.Issuer.CommonName
 916	}
 917
 918	var domains []string
 919	if len(cert.DNSNames) > 0 {
 920		websiteSSL.PrimaryDomain = cert.DNSNames[0]
 921		domains = cert.DNSNames[1:]
 922	}
 923	if len(cert.IPAddresses) > 0 {
 924		if websiteSSL.PrimaryDomain == "" {
 925			websiteSSL.PrimaryDomain = cert.IPAddresses[0].String()
 926			for _, ip := range cert.IPAddresses[1:] {
 927				domains = append(domains, ip.String())
 928			}
 929		} else {
 930			for _, ip := range cert.IPAddresses {
 931				domains = append(domains, ip.String())
 932			}
 933		}
 934	}
 935	websiteSSL.Domains = strings.Join(domains, ",")
 936
 937	if websiteSSL.ID > 0 {
 938		if err := UpdateSSLConfig(*websiteSSL); err != nil {
 939			return err
 940		}
 941		if err := websiteSSLRepo.Save(websiteSSL); err != nil {
 942			return err
 943		}
 944		return pushSSLToNodeWithNewLogger(websiteSSL)
 945	}
 946	if err := websiteSSLRepo.Create(context.Background(), websiteSSL); err != nil {
 947		return err
 948	}
 949	return pushSSLToNodeWithNewLogger(websiteSSL)

 998		}
