  54func handleRuntimeRecover(runtime *model.Runtime, recoverFile string, isRollback bool, secret string) error {
  55	isOk := false
  56	if !isRollback {
  57		rollbackFile := path.Join(global.Dir.TmpDir, fmt.Sprintf("runtime/%s_%s.tar.gz", runtime.Name, time.Now().Format(constant.DateTimeSlimLayout)))
  58		if err := handleRuntimeBackup(runtime, path.Dir(rollbackFile), path.Base(rollbackFile), "", secret); err != nil {
  59			return fmt.Errorf("backup runtime %s for rollback before recover failed, err: %v", runtime.Name, err)
  60		}
  61		defer func() {
  62			if !isOk {
  63				global.LOG.Info("recover failed, start to rollback now")
  64				if err := handleRuntimeRecover(runtime, rollbackFile, true, ""); err != nil {
  65					global.LOG.Errorf("rollback runtime %s from %s failed, err: %v", runtime.Name, rollbackFile, err)
  66					return
  67				}
  68				global.LOG.Infof("rollback runtime %s from %s successful", runtime.Name, rollbackFile)
  69				_ = os.RemoveAll(rollbackFile)
  70			} else {
  71				_ = os.RemoveAll(rollbackFile)
  72			}
  73		}()
  74	}
  75	fileOp := files.NewFileOp()
  76	if err := fileOp.TarGzExtractPro(recoverFile, path.Dir(recoverFile), secret); err != nil {
  77		return err

 283func parseGoroutineDump(reader io.Reader, maxGroups int) ([]dto.RuntimeGoroutineGroup, bool) {
 284	type groupValue struct {
 285		state string
 286		top   string
 287		stack []string
 288		count int
 289	}
 290	groups := make(map[string]*groupValue)
 291	truncated := false
 292	scanner := bufio.NewScanner(reader)
 293	scanner.Buffer(make([]byte, 64*1024), 4*1024*1024)
 294	var block []string
 295	flush := func() {
 296		if len(block) == 0 {
 297			return
 298		}
 299		matches := goroutineHeaderPattern.FindStringSubmatch(block[0])
 300		state := "unknown"
 301		if len(matches) == 2 {
 302			state = matches[1]
 303		}
 304		stack := append([]string(nil), block[1:]...)
 305		if len(stack) > 40 {
 306			stack = stack[:40]
 307		}
 308		top := "runtime"
 309		functionLines := make([]string, 0, len(stack)/2)
 310		for _, line := range stack {
 311			trimmed := strings.TrimSpace(line)
 312			if trimmed == "" || strings.HasPrefix(trimmed, "/") || strings.HasPrefix(trimmed, "created by ") {
 313				continue
 314			}
 315			normalized := goroutineArgPattern.ReplaceAllString(trimmed, "0x…")
 316			functionLines = append(functionLines, normalized)
 317			if top == "runtime" {
 318				top = strings.Split(normalized, "(")[0]
 319			}
 320		}
 321		signature := state + "\n" + strings.Join(functionLines, "\n")
 322		if existing, ok := groups[signature]; ok {
 323			existing.count++
 324			return
 325		}

 614func handleCompose(env gotenv.Env, composeContent []byte, create request.RuntimeCreate, projectDir string) (composeByte []byte, err error) {
 615	existMap := make(map[string]interface{})
 616	composeMap := make(map[string]interface{})
 617	if err = yaml.Unmarshal(composeContent, &composeMap); err != nil {
 618		return
 619	}
 620	services, serviceValid := composeMap["services"].(map[string]interface{})
 621	if !serviceValid {
 622		err = buserr.New("ErrFileParse")
 623		return
 624	}
 625	serviceName := ""
 626	serviceValue := make(map[string]interface{})
 627	for name, service := range services {
 628		serviceName = name
 629		serviceValue = service.(map[string]interface{})
 630		delete(serviceValue, "ports")
 631		if len(create.ExposedPorts) > 0 {
 632			var ports []interface{}
 633			for i, port := range create.ExposedPorts {
 634				containerPortStr, hostPortStr, hostIPStr, protocolStr := composePortEnvKeys(i)
 635				existMap[containerPortStr] = struct{}{}
 636				existMap[hostPortStr] = struct{}{}
 637				existMap[hostIPStr] = struct{}{}
 638				existMap[protocolStr] = struct{}{}
 639				ports = append(ports, formatComposePortMapping(hostIPStr, hostPortStr, containerPortStr, port.Protocol))
 640				create.Params[containerPortStr] = port.ContainerPort
 641				create.Params[hostPortStr] = port.HostPort
 642				create.Params[hostIPStr] = port.HostIP
 643				create.Params[protocolStr] = normalizeComposeProtocol(port.Protocol)
 644			}
 645			if create.Type == constant.RuntimePHP {
 646				ports = append(ports, "127.0.0.1:${PANEL_APP_PORT_HTTP}:9000")
 647			}
 648			serviceValue["ports"] = ports
 649		} else {
 650			if create.Type == constant.RuntimePHP {
 651				serviceValue["ports"] = []interface{}{"127.0.0.1:${PANEL_APP_PORT_HTTP}:9000"}
 652			}
 653		}
 654		var environments []interface{}
 655		for _, e := range create.Environments {
 656			environments = append(environments, fmt.Sprintf("%s=%s", e.Key, e.Value))
 657		}
 658		delete(serviceValue, "environment")
 659		if len(environments) > 0 {
 660			serviceValue["environment"] = environments
 661		}
 662		var volumes []interface{}
 663		defaultVolumes := make(map[string]string)
 664		switch create.Type {
 665		case constant.RuntimeNode, constant.RuntimeJava, constant.RuntimePython, constant.RuntimeDotNet:
 666			defaultVolumes = constant.RuntimeDefaultVolumes
 667		case constant.RuntimeGo:
 668			defaultVolumes = constant.GoDefaultVolumes
 669		case constant.RuntimePHP:
 670			defaultVolumes = constant.PHPDefaultVolumes
 671		}
 672		for k, v := range defaultVolumes {
 673			volumes = append(volumes, fmt.Sprintf("%s:%s", k, v))
 674		}
 675		for _, volume := range create.Volumes {
 676			volumes = append(volumes, formatComposeVolume(volume.Source, volume.Target, volume.Mode))
 677		}
 678
 679		var extraHosts []interface{}
 680		for _, host := range create.ExtraHosts {
 681			extraHosts = append(extraHosts, fmt.Sprintf("%s:%s", host.Hostname, host.IP))
 682		}
 683		delete(serviceValue, "extra_hosts")
 684		if len(extraHosts) > 0 {
 685			serviceValue["extra_hosts"] = extraHosts
 686		}
 687
 688		serviceValue["volumes"] = volumes
 689		break
 690	}
 691	for k := range env {
 692		if isComposePortEnvKey(k) {
 693			if _, ok := existMap[k]; !ok {
 694				delete(env, k)
 695			}
 696		}
 697	}
 698
 699	services[serviceName] = serviceValue
 700	composeMap["services"] = services
 701	composeByte, err = yaml.Marshal(composeMap)
 702	if err != nil {
 703		return
 704	}
 705	fileOp := files.NewFileOp()
 706	_ = fileOp.SaveFile(path.Join(projectDir, "docker-compose.yml"), string(composeByte), constant.DirPerm)
 707	return
 708}

 804	if err != nil {
 805		return err
 806	}
 807	defer file.Close()

 809	targetLine2 := `zend_extension="` + extensionName + `"`

 812	if err != nil {

 814	}
 815	defer tempFile.Close()
 816
 817	scanner := bufio.NewScanner(file)
 818	writer := bufio.NewWriter(tempFile)

 826			}
 827		}
 828	}
 829

 831		return err
