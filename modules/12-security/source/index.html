 342func handleFirewallRuleError(c *gin.Context, err error) {
 343	switch {
 344	case errors.Is(err, filter.ErrLockoutRisk), errors.Is(err, filter.ErrProtectedRule):
 345		helper.ErrorWithBusinessCode(c, http.StatusBadRequest, "FW_LOCKOUT_RISK", "ErrInvalidParams", err)
 346	case errors.Is(err, filter.ErrRuleStale):
 347		helper.ErrorWithBusinessCode(c, http.StatusConflict, "FW_RULE_STALE", "ErrInvalidParams", err)
 348	case errors.Is(err, repo.ErrFirewallRuleRevisionConflict):
 349		helper.ErrorWithBusinessCode(c, http.StatusConflict, "FW_RULE_REVISION_CONFLICT", "ErrInvalidParams", err)
 350	case errors.Is(err, filter.ErrRuleCheckRequired):
 351		helper.ErrorWithBusinessCode(c, http.StatusConflict, "FW_RULE_CHECK_REQUIRED", "ErrInvalidParams", err)
 352	case errors.Is(err, filter.ErrUnsupportedScope), errors.Is(err, filter.ErrInvalidScope),
 353		errors.Is(err, filter.ErrProviderUnavailable), errors.Is(err, filter.ErrAdapterUnavailable):
 354		helper.ErrorWithBusinessCode(c, http.StatusBadRequest, "FW_SCOPE_UNSUPPORTED", "ErrInvalidParams", err)
 355	case errors.Is(err, filter.ErrInvalidRule), errors.Is(err, filter.ErrRuleOperation),
 356		errors.Is(err, repo.ErrFirewallPersistenceInvalid):
 357		helper.ErrorWithBusinessCode(c, http.StatusBadRequest, "FW_RULE_UNSUPPORTED", "ErrInvalidParams", err)
 358	case errors.Is(err, filter.ErrVerificationFailed):
 359		helper.ErrorWithBusinessCode(c, http.StatusInternalServerError, "FW_VERIFY_FAILED", "ErrInternalServer", err)
 360	default:
 361		helper.ErrorWithBusinessCode(c, http.StatusInternalServerError, "FW_APPLY_FAILED", "ErrInternalServer", err)
 362	}
 363}

 464func normalizeGuardPolicy(family, hostIP string, hostPort uint16, protocol, mode string, sources []string) (normalizedDockerGuardPolicy, []string, error) {
 465	family, hostIP, protocol, mode = strings.ToLower(strings.TrimSpace(family)), strings.TrimSpace(hostIP), strings.ToLower(strings.TrimSpace(protocol)), strings.ToLower(strings.TrimSpace(mode))
 466	if hostPort == 0 || (protocol != "tcp" && protocol != "udp") || (family != docker_guard.FamilyIPv4 && family != docker_guard.FamilyIPv6) || (mode != docker_guard.ModeAll && mode != docker_guard.ModeSources && mode != docker_guard.ModeAllow) {
 467		return normalizedDockerGuardPolicy{}, nil, fmt.Errorf("%w: invalid policy fields", ErrDockerGuardInvalid)
 468	}
 469	addr, err := netip.ParseAddr(hostIP)
 470	if err != nil || (family == docker_guard.FamilyIPv4) != addr.Is4() {
 471		return normalizedDockerGuardPolicy{}, nil, fmt.Errorf("%w: host IP does not match address family", ErrDockerGuardInvalid)
 472	}
 473	normalizedSources := make([]string, 0, len(sources))
 474	seen := map[string]struct{}{}
 475	for _, source := range sources {
 476		source = strings.TrimSpace(source)
 477		if source == "" {
 478			continue
 479		}
 480		prefix, err := netip.ParsePrefix(source)
 481		if err != nil {
 482			if sourceAddr, addrErr := netip.ParseAddr(source); addrErr == nil {
 483				bits := 128
 484				if sourceAddr.Is4() {
 485					bits = 32
 486				}
 487				prefix = netip.PrefixFrom(sourceAddr, bits)
 488			} else {
 489				return normalizedDockerGuardPolicy{}, nil, fmt.Errorf("%w: invalid source address %q", ErrDockerGuardInvalid, source)
 490			}
 491		}
 492		if (family == docker_guard.FamilyIPv4) != prefix.Addr().Is4() {
 493			return normalizedDockerGuardPolicy{}, nil, fmt.Errorf("%w: source %q does not match address family", ErrDockerGuardInvalid, source)
 494		}
 495		canonical := prefix.Masked().String()
 496		if _, ok := seen[canonical]; !ok {
 497			seen[canonical] = struct{}{}
 498			normalizedSources = append(normalizedSources, canonical)
 499		}
 500	}
 501	if mode == docker_guard.ModeSources && len(normalizedSources) == 0 {
 502		return normalizedDockerGuardPolicy{}, nil, fmt.Errorf("%w: deny_sources requires at least one source", ErrDockerGuardInvalid)
 503	}
 504	if mode == docker_guard.ModeAll {
 505		normalizedSources = []string{}
 506	}
 507	sort.Strings(normalizedSources)
 508	return normalizedDockerGuardPolicy{Family: family, HostIP: hostIP, HostPort: hostPort, Protocol: protocol, Mode: mode}, normalizedSources, nil
 509}

 529func groupDockerGuardContainers(endpoints []dto.DockerPortGuardEndpoint) []dto.DockerPortGuardContainer {
 530	containers := make(map[string]*dto.DockerPortGuardContainer)
 531	order := make([]string, 0)
 532	for _, endpoint := range endpoints {
 533		key := endpoint.ContainerID
 534		if key == "" {
 535			key = "__orphan__"
 536		}
 537		container, ok := containers[key]
 538		if !ok {
 539			container = &dto.DockerPortGuardContainer{
 540				Key: key, Name: endpoint.ContainerName, Compose: endpoint.Compose,
 541				Application: endpoint.Application, Endpoints: []dto.DockerPortGuardEndpoint{},
 542			}
 543			containers[key] = container
 544			order = append(order, key)
 545		}
 546		container.Endpoints = append(container.Endpoints, endpoint)
 547	}
 548	sort.Slice(order, func(i, j int) bool {
 549		return containers[order[i]].Name < containers[order[j]].Name
 550	})
 551
 552	result := make([]dto.DockerPortGuardContainer, 0, len(order))
 553	for _, key := range order {
 554		container := containers[key]
 555		items := make([]docker.PortRangeItem, 0, len(container.Endpoints))
 556		for i, endpoint := range container.Endpoints {
 557			sources := append([]string(nil), endpoint.Sources...)
 558			sort.Strings(sources)
 559			policyKey := fmt.Sprintf("%t|%s|%s|%t|%s", endpoint.PolicyUUID != "", endpoint.Mode, strings.Join(sources, ","), endpoint.Effective, endpoint.Description)
 560			items = append(items, docker.PortRangeItem{
 561				Key:        endpoint.Family + "|" + endpoint.HostIP + "|" + endpoint.Protocol + "|" + policyKey,
 562				PublicPort: endpoint.HostPort, PrivatePort: endpoint.ContainerPort,
 563				HasPrivatePort: endpoint.ContainerPort != 0, Position: i,
 564			})
 565		}
 566		container.PortGroups = make([]dto.DockerPortGuardPortGroup, 0, len(items))
 567		for _, portRange := range docker.MergePortRanges(items) {
 568			start := container.Endpoints[portRange.Start.Position]
 569			address := start.HostIP
 570			if strings.Contains(address, ":") {
 571				address = "[" + address + "]"
 572			}
 573			ports := fmt.Sprintf("%d", portRange.Start.PublicPort)
 574			if portRange.Start.PublicPort != portRange.End.PublicPort {
 575				ports = fmt.Sprintf("%d-%d", portRange.Start.PublicPort, portRange.End.PublicPort)
 576			}
 577			container.PortGroups = append(container.PortGroups, dto.DockerPortGuardPortGroup{
 578				Key:   fmt.Sprintf("%s|%d-%d", portRange.Start.Key, portRange.Start.PublicPort, portRange.End.PublicPort),
 579				Label: fmt.Sprintf("%s:%s/%s", address, ports, start.Protocol), Endpoint: start,
 580				Endpoints: func() []dto.DockerPortGuardEndpoint {
 581					members := make([]dto.DockerPortGuardEndpoint, 0, len(portRange.Items))
 582					for _, item := range portRange.Items {
 583						members = append(members, container.Endpoints[item.Position])
 584					}
 585					return members
 586				}(),
 587			})
 588		}
 589		result = append(result, *container)
 590	}
 591	return result
 592}

 416	scope := rule.Scope

 422	}
 423	if scope.Direction == "" {
 424		scope.Direction = filter.DirectionInput
 425	}

 440		}
 441	case filter.ProviderUFW:
 442		if scope.Chain == "" {
 443			scope.Chain = filter.UFWInputChain
 444		}
 445	}
 446	rule.Scope = scope
 447	return rule

1381	after, err := filter.NormalizeRule(requestedRule)
1382	if err != nil {
1383		return preparedManagedUpdate{}, err
1384	}
1385	after.UUID = stored.UUID
1386	after, err = runtime.Prepare(after)
1387	if err != nil {
1388		return preparedManagedUpdate{}, err
1389	}
1390	if err := runtime.CheckRule(ctx, after); err != nil {
1391		return preparedManagedUpdate{}, err
1392	}
1393	if after.Scope.Key() != before.Rule.Scope.Key() {
1394		return preparedManagedUpdate{}, fmt.Errorf("%w: managed rule scope cannot be changed", filter.ErrUnsupportedScope)
1395	}
1396	if after.NativeKind != before.Rule.NativeKind {
1397		return preparedManagedUpdate{}, fmt.Errorf("%w: native rule conversion requires an explicit workflow", filter.ErrUnsupportedScope)
1398	}
1399	capabilities, err := runtime.Capabilities(ctx)
1400	if err != nil {
1401		return preparedManagedUpdate{}, err
1402	}
1403	if capabilities.ExplicitPosition || capabilities.OwnedChains {
1404		if observed.Locator.Position == nil {
1405			return preparedManagedUpdate{}, fmt.Errorf("%w: managed rule has no positional locator", filter.ErrInvalidRule)
1406		}
1407		currentPosition := int64(*observed.Locator.Position)
1408		if after.OrderIndex == nil {
1409			after.OrderIndex = &currentPosition
1410		} else if *after.OrderIndex != currentPosition {
1411			if err := validatePositionTarget(ctx, runtime, snapshot, before.Rule, *after.OrderIndex); err != nil {
1412				return preparedManagedUpdate{}, err
1413			}
1414		}
1415	}
1416	protectedPorts, err := s.loadProtectedPorts()
1417	if err != nil {
1418		return preparedManagedUpdate{}, err
1419	}
1420	if err := filter.GuardMutation(snapshot, observed, after, clientIP, protectedPorts...); err != nil {
1421		return preparedManagedUpdate{}, err
1422	}
1423	return preparedManagedUpdate{
1424		Stored: stored, Before: before, After: after, Snapshot: snapshot, Observed: observed, Runtime: runtime,
1425	}, nil
1426}

1428func (s *FirewallService) loadManagedMutation(
1429	ctx context.Context,
1430	ruleUUID string,
1431) (model.FirewallRule, filter.DesiredRule, filter.Snapshot, filter.ObservedRule, *firewallRuleRuntime, error) {
1432	stored, err := s.rules.GetByUUID(ctx, ruleUUID)
1433	if err != nil {
1434		return model.FirewallRule{}, filter.DesiredRule{}, filter.Snapshot{}, filter.ObservedRule{}, nil, err
1435	}
1436	if stored.Origin != constant.FirewallRuleOriginCreated && stored.Origin != constant.FirewallRuleOriginAdopted {
1437		return model.FirewallRule{}, filter.DesiredRule{}, filter.Snapshot{}, filter.ObservedRule{}, nil,
1438			fmt.Errorf("%w: only created or adopted rules can be changed", filter.ErrInvalidRule)
1439	}
1440	desired, err := desiredFirewallRuleFromModel(stored)
1441	if err != nil {
1442		return model.FirewallRule{}, filter.DesiredRule{}, filter.Snapshot{}, filter.ObservedRule{}, nil, err
1443	}
1444	runtime, err := s.resolveRuntime(ctx, desired.Rule.Scope.Provider)
1445	if err != nil {
1446		return model.FirewallRule{}, filter.DesiredRule{}, filter.Snapshot{}, filter.ObservedRule{}, nil, err
1447	}
1448	snapshot, err := runtime.ObserveMutation(ctx, desired.Rule.Scope)
1449	if err != nil {
1450		return model.FirewallRule{}, filter.DesiredRule{}, filter.Snapshot{}, filter.ObservedRule{}, nil, err
1451	}
1452	observed, err := filter.ManagedObserved(snapshot, desired)
1453	if err != nil {
1454		return model.FirewallRule{}, filter.DesiredRule{}, filter.Snapshot{}, filter.ObservedRule{}, nil, err
1455	}
1456	return stored, desired, snapshot, observed, runtime, nil
1457}

1459func (s *FirewallService) executeManagedMutation(ctx context.Context, request managedMutationRequest) error {
1460	before, after := request.Before, request.After
1461	semantic, err := model.FirewallRuleFromDomain(after)
1462	if err != nil {
1463		return err
1464	}
1465	if err := s.ensureFirewallRuleIdentityAvailable(
1466		ctx, semantic.ScopeKey, semantic.RuleKey, "", request.Stored.UUID,
1467	); err != nil {
1468		return err
1469	}
1470	appendRule, restoreAtEnd := false, false
1471	if after.Scope.Provider == filter.ProviderUFW && request.AdapterOperation == filter.ChangeUpdate {
1472		maxPosition, maxErr := maxPositionForRule(ctx, request.Runtime, request.Snapshot, after)
1473		if maxErr != nil {
1474			return maxErr
1475		}
1476		appendRule = after.OrderIndex != nil && *after.OrderIndex == maxPosition
1477		restoreAtEnd = request.Locator.Position != nil && int64(*request.Locator.Position) == maxPosition
1478	}
1479	backendPlan, verification, err := request.Runtime.Execute(ctx, request.Snapshot, []filter.DesiredChange{{
1480		Operation:    request.AdapterOperation,
1481		Before:       &before,
1482		After:        &after,
1483		Locator:      &request.Locator,
1484		Append:       appendRule,
1485		RestoreAtEnd: restoreAtEnd,
1486	}})
1487	if err != nil {
1488		return err
1489	}
1490	if !verification.Matched {
1491		return filter.ErrVerificationFailed
1492	}
1493	observed, err := filter.FindCommittedObserved(verification.Snapshot, request.After, backendPlan)
1494	if err != nil {
1495		return rollbackFirewallPlan(ctx, request.Runtime, backendPlan, err)
1496	}
1497	updates, err := firewallRuleSemanticUpdates(request.After)
1498	if err != nil {
1499		return rollbackFirewallPlan(ctx, request.Runtime, backendPlan, err)
1500	}
1501	applied, err := appliedFirewallRuleUpdates(observed)
1502	if err != nil {
1503		return rollbackFirewallPlan(ctx, request.Runtime, backendPlan, err)
1504	}
1505	for key, value := range applied {
1506		updates[key] = value
1507	}
1508	matchKey, _ := updates["match_key"].(string)
1509	if err := s.ensureFirewallRuleIdentityAvailable(
1510		ctx, semantic.ScopeKey, "", matchKey, request.Stored.UUID,
1511	); err != nil {
1512		return rollbackFirewallPlan(ctx, request.Runtime, backendPlan, err)
1513	}
1514	if err := s.rules.UpdateWithRevision(ctx, request.Stored.UUID, request.Stored.Revision, updates); err != nil {
1515		return rollbackFirewallPlan(ctx, request.Runtime, backendPlan, err)
1516	}
1517	return nil
1518}

1521	ctx context.Context,
1522	scopeKey string,
1523	ruleKey string,
1524	matchKey string,

1526) error {

1538		if matchKey != "" && candidate.MatchKey == matchKey {
1539			return fmt.Errorf("%w: firewall rule instance is already managed", filter.ErrRuleOperation)

1541	}
1542	return nil

1601func (s *FirewallService) prepareSystemPortCreate(ctx context.Context, port dto.FirewallSystemPort) (*dto.FirewallRuleCreateItem, error) {
1602	provider, err := s.selectedProvider(ctx)
1603	if err != nil {
1604		return nil, err
1605	}

1614		CheckFlag:  check.CheckFlag,

1619	case filter.CheckClassificationNone:
1620		create.Action = filter.CheckActionCreate

1627		} else {
1628			create.Action = filter.CheckActionSelectAdopt
1629		}
1630		create.AdoptInstanceKey = check.Candidates[0].InstanceKey
1631	case filter.CheckClassificationCovered:
1632		create.Action = filter.CheckActionCreateAnyway
1633	case filter.CheckClassificationExactManaged:
1634		return nil, nil

1636		if len(check.Candidates) > 0 {
1637			return nil, nil
1638		}
1639		return nil, fmt.Errorf("%w: protected accepted port %s", filter.ErrProtectedRule, port.Port)
1640	default:
1641		return nil, fmt.Errorf("cannot manage accepted port %s/%s: %s", port.Port, port.Protocol, check.Reason)
1642	}
1643	return &create, nil
1644}

1739func systemPortRule(provider filter.Provider, port dto.FirewallSystemPort) filter.FirewallRule {
1740	scope := filter.Scope{Provider: provider, Direction: filter.DirectionInput}
1741	family := filter.Family(strings.ToLower(strings.TrimSpace(port.Family)))
1742	switch provider {
1743	case filter.ProviderIptables, filter.ProviderNftables:
1744		if family != filter.FamilyIPv6 {
1745			family = filter.FamilyIPv4
1746		}
1747		scope.Family = family
1748		scope.Table = "filter"
1749	case filter.ProviderFirewalld:
1750		if family != filter.FamilyIPv4 && family != filter.FamilyIPv6 {
1751			family = filter.FamilyInet
1752		}
1753		scope.Family = family
1754		scope.Zone = filter.FirewalldInputZone
1755	case filter.ProviderUFW:
1756		if family != filter.FamilyIPv6 {
1757			family = filter.FamilyIPv4
1758		}
1759		scope.Family = family
1760	}
1761	return filter.FirewallRule{
1762		Scope: scope, Protocol: port.Protocol, DestinationPort: port.Port,
1763		Action: filter.ActionAccept, Description: "1Panel managed accepted port",
1764	}

1895func desiredFirewallRuleFromModel(stored model.FirewallRule) (filter.DesiredRule, error) {
1896	connectionStates := make([]string, 0)
1897	if stored.ConnectionStates != "" {
1898		connectionStates = strings.Split(stored.ConnectionStates, ",")
1899	}
1900	scope := filter.Scope{
1901		Provider: filter.Provider(stored.Provider), Family: filter.Family(stored.Family), Direction: filter.DirectionInput,
1902	}
1903	switch scope.Provider {
1904	case filter.ProviderIptables, filter.ProviderNftables:
1905		scope.Table = "filter"
1906		scope.Chain = stored.Location
1907	case filter.ProviderFirewalld:
1908		scope.Zone = stored.Location
1909	case filter.ProviderUFW:
1910		scope.Chain = stored.Location
1911	}
1912	rule := filter.FirewallRule{
1913		UUID:               stored.UUID,
1914		Scope:              scope,
1915		NativeKind:         filter.NativeKind(stored.NativeKind),
1916		Protocol:           stored.Protocol,
1917		SourceAddress:      stored.SourceAddress,
1918		SourcePort:         stored.SourcePort,
1919		DestinationAddress: stored.DestinationAddress,
1920		DestinationPort:    stored.DestinationPort,
1921		Interface:          stored.Interface,
1922		ConnectionStates:   connectionStates,
1923		Action:             filter.Action(stored.Action),
1924		Priority:           stored.Priority,
1925		OrderIndex:         stored.OrderIndex,
1926		OrderBucket:        stored.OrderBucket,
1927		Description:        stored.Description,
1928	}
1929	normalized, err := filter.NormalizeRule(rule)
1930	if err != nil {
1931		return filter.DesiredRule{}, fmt.Errorf("normalize firewall rule %q payload: %w", stored.UUID, err)
1932	}
1933	if normalized.Scope.Key() != stored.ScopeKey {
1934		return filter.DesiredRule{}, fmt.Errorf("%w: stored firewall rule %q scope key does not match payload", filter.ErrInvalidRule, stored.UUID)
1935	}
1936	ruleKey, err := filter.RuleKey(normalized)
1937	if err != nil {
1938		return filter.DesiredRule{}, err
1939	}
1940	if ruleKey != stored.RuleKey {
1941		return filter.DesiredRule{}, fmt.Errorf("%w: stored firewall rule %q key does not match payload", filter.ErrInvalidRule, stored.UUID)
1942	}
1943	marker, observedInstanceKey := firewallRuleMatchValues(stored.MatchKey)
1944	return filter.DesiredRule{
1945		UUID:                stored.UUID,
1946		Rule:                normalized,
1947		RuleKey:             stored.RuleKey,
1948		Origin:              filter.RuleOrigin(stored.Origin),
1949		Marker:              marker,
1950		ObservedInstanceKey: observedInstanceKey,
1951	}, nil
1952}
