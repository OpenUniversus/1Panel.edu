 487	form, err := c.MultipartForm()
 488	if err != nil {
 489		helper.BadRequest(c, err)
 490		return
 491	}
 492	uploadFiles := form.File["file"]
 493	paths := form.Value["path"]
 494
 495	overwrite := true
 496	if ow, ok := form.Value["overwrite"]; ok {
 497		if len(ow) != 0 {
 498			parseBool, _ := strconv.ParseBool(ow[0])
 499			overwrite = parseBool
 500		}
 501	}
 502
 503	if len(paths) == 0 || !strings.Contains(paths[0], "/") {
 504		helper.BadRequest(c, errors.New("error paths in request"))
 505		return
 506	}
 507	dir := path.Clean(paths[0])
 508
 509	_, err = os.Stat(dir)
 510	if err != nil && os.IsNotExist(err) {
 511		mode, err := files.GetParentMode(dir)
 512		if err != nil {
 513			helper.InternalServer(c, err)
 514			return
 515		}
 516		if err = os.MkdirAll(dir, mode); err != nil {
 517			helper.BadRequest(c, fmt.Errorf("mkdir %s failed, err: %v", dir, err))
 518			return
 519		}
 520	}
 521	info, err := os.Stat(dir)
 522	if err != nil {
 523		helper.InternalServer(c, err)
 524		return
 525	}
 526	mode := info.Mode()
 527
 528	fileOp := files.NewFileOp()
 529	stat, ok := info.Sys().(*syscall.Stat_t)
 530	uid, gid := -1, -1
 531	if ok {
 532		uid, gid = int(stat.Uid), int(stat.Gid)
 533	}
 534	success := 0
 535	failures := make(buserr.MultiErr)
 536	for _, file := range uploadFiles {
 537		dstFilename := path.Join(paths[0], file.Filename)
 538		dstDir := path.Dir(dstFilename)
 539		if !fileOp.Stat(dstDir) {
 540			if err = fileOp.CreateDir(dstDir, mode); err != nil {
 541				e := fmt.Errorf("create dir [%s] failed, err: %v", path.Dir(dstFilename), err)
 542				failures[file.Filename] = e
 543				global.LOG.Error(e)
 544				continue
 545			}
 546			_ = os.Chown(dstDir, uid, gid)
 547		}
 548		dstDirMode := mode
 549		dstFileMode := mode.Perm()
 550		if dstDirInfo, err := os.Stat(dstDir); err == nil {
 551			dstDirMode = dstDirInfo.Mode()
 552			dstFileMode = dstDirInfo.Mode().Perm()
 553		}
 554		tmpFilename := dstFilename + ".tmp"
 555		if err := c.SaveUploadedFile(file, tmpFilename, dstDirMode); err != nil {
 556			_ = os.Remove(tmpFilename)
 557			e := fmt.Errorf("upload [%s] file failed, err: %v", file.Filename, err)
 558			failures[file.Filename] = e
 559			global.LOG.Error(e)
 560			continue
 561		}
 562		dstInfo, statErr := os.Stat(dstFilename)
 563		err = finalizeUploadedFile(tmpFilename, dstFilename, overwrite)
 564		if err != nil {
 565			_ = os.Remove(tmpFilename)
 566			e := fmt.Errorf("upload [%s] file failed, err: %v", file.Filename, err)
 567			failures[file.Filename] = e
 568			global.LOG.Error(e)
 569			continue
 570		}
 571		if statErr == nil {
 572			_ = os.Chmod(dstFilename, dstInfo.Mode())
 573		} else {
 574			_ = os.Chmod(dstFilename, dstFileMode)
 575		}
 576		if uid != -1 && gid != -1 {
 577			_ = os.Chown(dstFilename, uid, gid)

1044func writeResumableUploadChunk(chunk resumableUploadChunk, chunkData []byte) error {
1045	unlock := lockChunkUpload(chunk.UploadID)
1046	defer unlock()
1047	if chunkUploadCancelled(chunk.UploadID) {
1048		return errChunkUploadCancelled
1049	}
1050	if chunk.UploadID == "" || filepath.Base(chunk.UploadID) != chunk.UploadID || strings.ContainsAny(chunk.UploadID, `/\`) {
1051		return invalidChunkUploadError("invalid upload ID")
1052	}
1053	if chunk.Filename == "" || filepath.Base(chunk.Filename) != chunk.Filename || strings.ContainsAny(chunk.Filename, `/\`) {
1054		return invalidChunkUploadError("invalid filename")
1055	}
1056	if strings.TrimSpace(chunk.DstDir) == "" {
1057		return invalidChunkUploadError("upload destination is required")
1058	}
1059	dstDir := filepath.Clean(strings.TrimSpace(chunk.DstDir))
1060
1061	if chunk.ChunkCount <= 0 || chunk.ChunkIndex < 0 || chunk.ChunkIndex >= chunk.ChunkCount {
1062		return invalidChunkUploadError("invalid chunk index")
1063	}
1064	if chunk.FileSize <= 0 || chunk.Offset < 0 || chunk.Offset > chunk.FileSize {
1065		return invalidChunkUploadError("invalid upload offset")
1066	}
1067	chunkEnd := chunk.Offset + int64(len(chunkData))
1068	if chunkEnd > chunk.FileSize {
1069		return invalidChunkUploadError("chunk exceeds file size")
1070	}
1071	if chunk.ChunkIndex+1 == chunk.ChunkCount {
1072		if chunkEnd != chunk.FileSize {
1073			return invalidChunkUploadError("final chunk does not match file size")
1074		}
1075	} else if chunkEnd >= chunk.FileSize {
1076		return invalidChunkUploadError("non-final chunk reaches file size")
1077	}
1078	if completed, ok := loadCompletedChunkUpload(chunk.UploadID); ok {
1079		if completed.dstDir == dstDir && completed.filename == chunk.Filename && completed.fileSize == chunk.FileSize {
1080			return nil
1081		}
1082		return invalidChunkUploadError("upload ID has already completed another file")
1083	}
1084	upload := completedChunkUpload{dstDir: dstDir, filename: chunk.Filename, fileSize: chunk.FileSize}
1085	if err := registerActiveChunkUpload(chunk.UploadID, upload); err != nil {
1086		return err
1087	}
1088
1089	mode, err := files.GetParentMode(dstDir)
1090	if err != nil {
1091		return err
1092	}
1093	if err = os.MkdirAll(dstDir, mode); err != nil {
1094		return err
1095	}
1096	dstDirInfo, err := os.Stat(dstDir)
1097	if err != nil {
1098		return err
1099	}
1100	if !dstDirInfo.IsDir() {
1101		return invalidChunkUploadError(fmt.Sprintf("upload destination [%s] is not a directory", dstDir))
1102	}
1103
1104	dstFile := filepath.Join(dstDir, chunk.Filename)
1105	partFile := resumableUploadPartPath(dstDir, chunk.UploadID)
1106	if dstFile == partFile {
1107		return invalidChunkUploadError("filename conflicts with upload temporary file")
1108	}
1109	fileMode := dstDirInfo.Mode().Perm()
1110	ownerInfo := dstDirInfo
1111	if dstInfo, statErr := os.Stat(dstFile); statErr == nil {
1112		if !chunk.Overwrite {
1113			if err := discardActiveChunkUpload(chunk.UploadID, partFile); err != nil {
1114				return errors.Join(os.ErrExist, err)
1115			}
1116			return os.ErrExist
1117		}
1118		fileMode = dstInfo.Mode().Perm()
1119		ownerInfo = dstInfo
1120	} else if !os.IsNotExist(statErr) {
1121		return statErr
1122	}
1123
1124	part, err := os.OpenFile(partFile, os.O_CREATE|os.O_RDWR, fileMode)
1125	if err != nil {
1126		return err
1127	}
1128	partClosed := false
1129	defer func() {
1130		if !partClosed {
1131			_ = part.Close()
1132		}
1133	}()
1134	if stat, statErr := part.Stat(); statErr != nil {
1135		return statErr
1136	} else if chunk.Offset > stat.Size() {
1137		return invalidChunkUploadError(fmt.Sprintf("unexpected upload offset %d, current size is %d", chunk.Offset, stat.Size()))
1138	} else if chunk.Offset < stat.Size() && chunkEnd > stat.Size() {
1139		if err = part.Truncate(chunk.Offset); err != nil {
1140			return err
1141		}
1142	}
1143	if _, err = part.WriteAt(chunkData, chunk.Offset); err != nil {
1144		return err
1145	}
1146
1147	if chunk.ChunkIndex+1 != chunk.ChunkCount {
1148		return nil
1149	}
1150	partInfo, err := part.Stat()
1151	if err != nil {
1152		return err
1153	}
1154	if partInfo.Size() != chunk.FileSize {
1155		return invalidChunkUploadError(fmt.Sprintf("uploaded file size mismatch: expected %d, got %d", chunk.FileSize, partInfo.Size()))
1156	}
1157	if err = part.Close(); err != nil {
1158		return err
1159	}
1160	partClosed = true
1161	if err = os.Chmod(partFile, fileMode); err != nil {
1162		return err
1163	}
1164	if stat, ok := ownerInfo.Sys().(*syscall.Stat_t); ok {
1165		if err = os.Chown(partFile, int(stat.Uid), int(stat.Gid)); err != nil {
1166			return err
1167		}
1168	}
1169	if chunkUploadCancelled(chunk.UploadID) {
1170		return errChunkUploadCancelled
1171	}
1172	if err = finalizeActiveChunkUpload(chunk.UploadID, partFile, dstFile, chunk.Overwrite); err != nil {
1173		return err
1174	}
1175	markChunkUploadCompleted(chunk.UploadID, upload)
1176	deleteActiveChunkUpload(chunk.UploadID)
1177	return nil
1178}

 210func (f *FileService) buildFileTree(node *response.FileTree, items []*files.FileInfo, op request.FileOption, level int) error {
 211	for _, v := range items {
 212		if shouldFilterPath(v.Path) {
 213			global.LOG.Infof("File Tree: Skipping %s due to filter\n", v.Path)
 214			continue
 215		}
 216		childNode := response.FileTree{
 217			ID:        common.GetUuid(),
 218			Name:      v.Name,
 219			Path:      v.Path,
 220			IsDir:     v.IsDir,
 221			Extension: v.Extension,
 222		}
 223		if level > 1 && v.IsDir {
 224			if err := f.buildChildNode(&childNode, v, op, level); err != nil {
 225				return err
 226			}
 227		}
 228
 229		node.Children = append(node.Children, childNode)
 230	}
 231	return nil
 232}

 234func (f *FileService) buildChildNode(childNode *response.FileTree, fileInfo *files.FileInfo, op request.FileOption, level int) error {

 241		}
 242		global.LOG.Errorf("File Tree: Skipping %s due to error: %s\n", fileInfo.Path, err.Error())
 243		return nil
 244	}

 606func copyDecompressEntryWithState(ctx context.Context, srcPath, dstPath string, state *decompressCopyState) (retErr error) {
 607	if err := ctx.Err(); err != nil {
 608		return err
 609	}
 610	info, err := os.Lstat(srcPath)
 611	if err != nil {
 612		return err
 613	}
 614
 615	if info.Mode()&os.ModeSymlink != 0 {
 616		if err := os.RemoveAll(dstPath); err != nil {
 617			return err
 618		}
 619		if err := os.MkdirAll(filepath.Dir(dstPath), constant.DirPerm); err != nil {
 620			return err
 621		}
 622		target, err := os.Readlink(srcPath)
 623		if err != nil {
 624			return err
 625		}
 626		if err := os.Symlink(target, dstPath); err != nil {
 627			return err
 628		}
 629		return applyDecompressOwnership(srcPath, dstPath)
 630	}
 631
 632	if info.IsDir() {
 633		dstInfo, err := os.Lstat(dstPath)
 634		keepExistingDir := err == nil && dstInfo.IsDir() && dstInfo.Mode()&os.ModeSymlink == 0
 635		if err != nil && !os.IsNotExist(err) {
 636			return err
 637		}
 638		if !keepExistingDir {
 639			if err := os.RemoveAll(dstPath); err != nil {
 640				return err
 641			}
 642			if err := os.MkdirAll(dstPath, info.Mode().Perm()); err != nil {
 643				return err
 644			}
 645			if err := applyDecompressOwnership(srcPath, dstPath); err != nil {
 646				return err
 647			}
 648			if err := os.Chmod(dstPath, info.Mode().Perm()); err != nil {
 649				return err
 650			}
 651		}
 652		entries, err := os.ReadDir(srcPath)
 653		if err != nil {
 654			return err
 655		}
 656		for _, entry := range entries {
 657			if err := copyDecompressEntryWithState(ctx, filepath.Join(srcPath, entry.Name()), filepath.Join(dstPath, entry.Name()), state); err != nil {
 658				return err
 659			}
 660		}
 661		if keepExistingDir {
 662			return nil
 663		}
 664		return os.Chtimes(dstPath, info.ModTime(), info.ModTime())
 665	}
 666
 667	dstInfo, err := os.Lstat(dstPath)
 668	keepExistingFile := err == nil && dstInfo.Mode().IsRegular()
 669	if err != nil && !os.IsNotExist(err) {
 670		return err
 671	}
 672	if !keepExistingFile {
 673		if err := os.RemoveAll(dstPath); err != nil {
 674			return err
 675		}
 676	}
 677	if err := os.MkdirAll(filepath.Dir(dstPath), constant.DirPerm); err != nil {
 678		return err
 679	}
 680	identity, isHardlink := decompressHardlinkIdentity(info)
 681	if !keepExistingFile && isHardlink {
 682		if existingPath, ok := state.hardlinks[identity]; ok {
 683			if err := os.Link(existingPath, dstPath); err != nil {
 684				return err
 685			}
 686			return os.Chtimes(dstPath, info.ModTime(), info.ModTime())
 687		}
 688	}
 689
 690	srcFile, err := os.Open(srcPath)
 691	if err != nil {
 692		return err
 693	}
 694	defer srcFile.Close()
 695
 696	dstFile, err := os.OpenFile(dstPath, os.O_CREATE|os.O_RDWR|os.O_TRUNC, info.Mode().Perm())
 697	if err != nil {
 698		return err
 699	}
 700	defer func() {
 701		if cerr := dstFile.Close(); cerr != nil && retErr == nil {
 702			retErr = cerr
 703		}
 704	}()
 705
 706	if _, err := io.Copy(dstFile, srcFile); err != nil {
 707		return err
 708	}
 709	if !keepExistingFile {
 710		if err := applyDecompressOwnership(srcPath, dstPath); err != nil {
 711			return err
 712		}
 713		if err := os.Chmod(dstPath, info.Mode().Perm()); err != nil {
 714			return err
 715		}
 716		if isHardlink {
 717			state.hardlinks[identity] = dstPath
 718		}
 719	}
 720	return os.Chtimes(dstPath, info.ModTime(), info.ModTime())
 721}

1046func readEditableFileHistoryContent(filePath string) ([]byte, os.FileMode, bool) {
1047	info, err := os.Lstat(filePath)
1048	if err != nil {
1049		return nil, 0640, false
1050	}
1051	mode := info.Mode()
1052	if mode.IsDir() || mode&os.ModeSymlink != 0 || !mode.IsRegular() || files.IsBlockDevice(mode) || info.Size() > fileHistorySnapshotMaxSize {
1053		return nil, mode, false
1054	}
1055	if files.IsBinaryPreviewFile(files.GetMimeType(filePath), filepath.Ext(info.Name())) {
1056		return nil, mode, false
1057	}
1058	file, err := os.Open(filePath)
1059	if err != nil {
1060		return nil, mode, false
1061	}
1062	defer file.Close()
1063
1064	headBuf := make([]byte, 1024)
1065	n, err := file.Read(headBuf)
1066	if err != nil && err != io.EOF {
1067		return nil, mode, false
1068	}
1069	if n > 0 && files.DetectBinary(headBuf[:n]) {
1070		return nil, mode, false
1071	}
1072	if _, err := file.Seek(0, 0); err != nil {
1073		return nil, mode, false
1074	}
1075	content, err := io.ReadAll(io.LimitReader(file, fileHistorySnapshotMaxSize+1))
1076	if err != nil || int64(len(content)) > fileHistorySnapshotMaxSize {
1077		return nil, mode, false
1078	}
1079	return content, mode, true
1080}

 366	}

 371			return nil, buserr.New("ErrFileShareInvalid")
 372		}
 373		return nil, err
 374	}
 375
 376	now := time.Now().Unix()
 377	if item.ExpiresUnix > 0 && now > item.ExpiresUnix {
 378		_ = fileShareRepo.Delete(repo.WithByID(item.ID))
 379		return nil, buserr.New("ErrFileShareExpired")
 380	}

 382		if subtle.ConstantTimeCompare([]byte(hashPassword(item.PasswordSalt, password)), []byte(item.PasswordHash)) != 1 {
 383			return nil, buserr.New("ErrFileSharePassword")
 384		}
 385	}
 386
 387	info, err := os.Stat(item.Path)
 388	if err != nil || info.IsDir() {
 389		_ = fileShareRepo.Delete(repo.WithByID(item.ID))
 390		return nil, buserr.New("ErrFileSharePath")
 391	}
