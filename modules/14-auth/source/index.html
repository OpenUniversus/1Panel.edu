 122func padding(plaintext []byte, blockSize int) []byte {
 123	padding := blockSize - len(plaintext)%blockSize
 124	padtext := bytes.Repeat([]byte{byte(padding)}, padding)
 125	return append(plaintext, padtext...)

 128func unPadding(origData []byte) ([]byte, error) {
 129	length := len(origData)
 130	if length == 0 {
 131		return nil, fmt.Errorf("invalid padding size")
 132	}
 133
 134	unpadding := int(origData[length-1])
 135	if unpadding == 0 || unpadding > length {
 136		return nil, fmt.Errorf("invalid padding")
 137	}
 138
 139	for i := 0; i < unpadding; i++ {
 140		if origData[length-1-i] != byte(unpadding) {
 141			return nil, fmt.Errorf("invalid padding")
 142		}
 143	}
 144
 145	return origData[:(length - unpadding)], nil

 148func aesEncryptWithSalt(key, plaintext []byte) ([]byte, error) {
 149	plaintext = padding(plaintext, aes.BlockSize)
 150	block, err := aes.NewCipher(key)
 151	if err != nil {
 152		return nil, err

 157		return nil, err
 158	}

 636		return []*net.IPNet{}, nil
 637	}
 638	cidrs := make([]*net.IPNet, 0, len(lines))
 639	for _, line := range lines {
 640		item := strings.TrimSpace(line)

 642			continue
 643		}
 644		if ip := net.ParseIP(item); ip != nil {
 645			if ip.To4() != nil {
 646				item += "/32"
 647			} else {
 648				item += "/128"
 649			}
 650		}
 651		_, ipNet, err := net.ParseCIDR(item)
 652		if err != nil {
 653			return nil, err
 654		}

 660func passkeyForwardedProto(value string) string {
 661	if value == "" {
 662		return ""
 663	}
 664	for _, item := range strings.Split(value, ",") {
 665		for _, part := range strings.Split(item, ";") {
 666			kv := strings.SplitN(strings.TrimSpace(part), "=", 2)
 667			if len(kv) != 2 {
 668				continue
 669			}
 670			if !strings.EqualFold(strings.TrimSpace(kv[0]), "proto") {
 671				continue
 672			}
 673			proto := strings.ToLower(strings.Trim(strings.TrimSpace(kv[1]), `"`))
 674			if proto == "http" || proto == "https" {
 675				return proto
 676			}
 677		}
 678	}
 679	return ""
 680}

 192	length := len(data)

 195	}
