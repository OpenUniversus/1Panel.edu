 231func (c *appSyncContext) syncAppIconsAndDetails() error {

 257		failFlag atomic.Bool
 258		workCh   = make(chan int, maxWorkers)
 259		resultCh = make(chan appWorkResult, maxWorkers)
 260		wg       sync.WaitGroup
 261	)
 262
 263	wg.Add(maxWorkers)
 264	for range maxWorkers {
 265		go func() {
 266			defer wg.Done()
 267			for idx := range workCh {
 268				result := c.processOneApp(workItems[idx])
 269				resultCh <- result
 270				if result.httpFailed {
 271					failFlag.Store(true)
 272				}
 273			}
 274		}()
 275	}
 276	go func() { wg.Wait(); close(resultCh) }()

 279	go func() {
 280		for i := range workItems {
 281			if failFlag.Load() {
 282				break
 283			}
 284			workCh <- i
 285			fed.Store(int32(i + 1))
 286		}
 287		close(workCh)
 288	}()
 289
 290	var (
 291		completed     int
 292		icon200Count  int
 293		icon304Count  int
 294		iconFailCount int
 295	)
 296	milestones := [4]int{totalWork / 4, totalWork / 2, totalWork * 3 / 4, totalWork}
 297	nextMS := 0
 298	checkProgress := func() {
 299		if nextMS < len(milestones) && completed >= milestones[nextMS] {
 300			c.task.LogWithProgress(i18n.GetMsgByKey("SyncAppDetail"), completed, totalWork)
 301			nextMS++
 302		}
 303	}
 304
 305	applyResult := func(result appWorkResult) {
 306		c.appsMap[result.appKey] = result.app
 307		if result.hadIcon {
 308			switch result.iconStatus {
 309			case http.StatusOK:
 310				icon200Count++
 311			case http.StatusNotModified:
 312				icon304Count++
 313			default:
 314				iconFailCount++
 315			}
 316		}
 317		completed++
 318		checkProgress()
 319	}
 320
 321	for result := range resultCh {
 322		applyResult(result)
 323	}
 324
 325	fedCount := int(fed.Load())
 326	if failFlag.Load() && fedCount < len(workItems) {
 327		global.LOG.Warnf("[AppStore] HTTP failure detected, falling back to serial for remaining %d apps", len(workItems)-fedCount)
 328		for i := fedCount; i < len(workItems); i++ {
 329			result := c.processOneApp(workItems[i])
 330			applyResult(result)
 331		}
 332	}
 333
 334	global.LOG.Infof("[AppStore] icon download completed - total: %d, success(200): %d, cached(304): %d, failed: %d",
 335		total, icon200Count, icon304Count, iconFailCount)
 336
 337	return nil

 340func (c *appSyncContext) downloadAppIcon(iconUrl, appKey, oldIcon string) (status int, iconField string) {
 341	iconFileName, existingEtag := appicon.ParseIconField(oldIcon)
 342
 343	reqHeaders := make(map[string]string)
 344	if existingEtag != "" && iconFileName != "" && appicon.IconFileExists(iconFileName) {
 345		reqHeaders["If-None-Match"] = existingEtag
 346	}
 347
 348	resp, err := req_helper.HandleRequestWithHeaders(&c.httpClient, iconUrl, http.MethodGet, constant.TimeOut20s, reqHeaders)
 349	if err != nil {
 350		global.LOG.Warnf("[AppStore] request icon failed url=%s, err=%v", iconUrl, err)
 351		return 0, ""
 352	}
 353
 354	if resp.StatusCode == http.StatusNotModified {
 355		return http.StatusNotModified, ""
 356	}
 357
 358	if resp.StatusCode != http.StatusOK {
 359		global.LOG.Warnf("[AppStore] download icon failed url=%s, status=%d", iconUrl, resp.StatusCode)
 360		return 0, ""
 361	}
 362
 363	if len(resp.Body) == 0 {
 364		global.LOG.Warnf("[AppStore] download icon empty body url=%s", iconUrl)
 365		return 0, ""
 366	}
 367
 368	if resp.Body[0] == '<' {
 369		global.LOG.Warnf("[AppStore] download icon got HTML response url=%s", iconUrl)
 370		return 0, ""
 371	}
 372
 373	contentType := resp.Header.Get("Content-Type")
 374	ct := strings.TrimSpace(strings.Split(contentType, ";")[0])
 375	if strings.ToLower(ct) != "image/png" {
 376		global.LOG.Warnf("[AppStore] unexpected icon content-type: %s, expected image/png, url=%s", ct, iconUrl)
 377	}
 378
 379	fileName, err := appicon.WriteIconFile(appKey, resp.Body)
 380	if err != nil {
 381		global.LOG.Warnf("[AppStore] write icon file failed appKey=%s, err=%v", appKey, err)
 382		return 0, ""
 383	}
 384
 385	newEtag := resp.Header.Get("ETag")
 386	iconField = appicon.BuildIconField(fileName, newEtag)
 387
 388	return http.StatusOK, iconField
 389}

 467func (u *CronjobService) LoadNextHandle(specStr string) ([]string, error) {
 468	spec := cron.NewParser(cron.Minute | cron.Hour | cron.Dom | cron.Month | cron.Dow)
 469	now := time.Now()
 470	var next [5]string
 471	if strings.HasPrefix(specStr, "@every ") {
 472		duration := time.Minute
 473		if strings.HasSuffix(specStr, "s") {
 474			duration = time.Second
 475		}
 476		interval := strings.ReplaceAll(specStr, "@every ", "")
 477		interval = strings.ReplaceAll(interval, "s", "")
 478		interval = strings.ReplaceAll(interval, "m", "")
 479		durationItem, err := strconv.Atoi(interval)
 480		if err != nil {
 481			return nil, err
 482		}
 483		for i := 0; i < 5; i++ {
 484			nextTime := now.Add(time.Duration(durationItem) * duration)
 485			next[i] = nextTime.Format(constant.DateTimeLayout)
 486			now = nextTime
 487		}
 488		return next[:], nil
 489	}
 490	sched, err := spec.Parse(specStr)
 491	if err != nil {
 492		return nil, err
 493	}
 494	for i := 0; i < 5; i++ {
 495		nextTime := sched.Next(now)
 496		next[i] = nextTime.Format(constant.DateTimeLayout)
 497		now = nextTime
 498	}
 499	return next[:], nil
 500}

 336			}

 338			outFile.Close()
 339			resp.Body.Close()
 340			taskItem.LogSuccess(i18n.Get("TaskSync") + " " + ipGroupFile)
 341		}

 450				}
