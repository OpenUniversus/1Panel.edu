 174			if err != nil {
 175				helper.BadRequest(c, err)
 176				return
 177			}
 178		}
 179		req.Password = ""
 180	}
 181
 182	upMap := map[string]interface{}{
 183		"name":              req.Name,
 184		"group_id":          req.GroupID,
 185		"addr":              req.Addr,
 186		"port":              req.Port,
 187		"user":              req.User,
 188		"auth_mode":         req.AuthMode,
 189		"remember_password": req.RememberPassword,
 190		"description":       req.Description,
 191	}
 192	if req.AuthMode == "password" {
 193		upMap["password"] = req.Password
 194		upMap["private_key"] = ""
 195		upMap["pass_phrase"] = ""
 196	} else {
 197		upMap["password"] = ""
 198		upMap["private_key"] = req.PrivateKey
 199		upMap["pass_phrase"] = req.PassPhrase
 200	}
 201	hostItem, err := hostService.Update(req.ID, upMap)

 205	}

 119func (u *SSHService) OperateSSH(operation string) error {
 120	serviceName, err := loadServiceName()
 121	if err != nil {
 122		return err
 123	}
 124	switch operation {
 125	case "start", "restart":
 126		if err := stopSSHSocketIfActive(serviceName); err != nil {
 127			return err
 128		}
 129	case "stop":
 130		if err := stopSSHSocketIfActive(serviceName); err != nil {
 131			return err
 132		}
 133	case "enable", "disable":
 134		if operation == "disable" {
 135			if err := disableSSHSocket(serviceName, false); err != nil {
 136				return err
 137			}
 138		} else {
 139			if err := disableSSHSocket(serviceName, true); err != nil {
 140				return err
 141			}
 142			if err := controller.Handle("enable", serviceName+".service"); err != nil {
 143				return fmt.Errorf("enable %s.service failed, err: %v", serviceName, err)
 144			}
 145			return nil
 146		}
 147		serviceName += ".service"
 148	}
 149
 150	if err := controller.Handle(operation, serviceName); err != nil {
 151		return fmt.Errorf("%s %s failed, err: %v", operation, serviceName, err)
 152	}
 153	return nil
 154}

 193func loadSSHSocketNames(serviceName string) []string {
 194	baseName := strings.TrimSuffix(serviceName, ".service")
 195	candidates := []string{baseName + ".socket"}
 196	switch baseName {
 197	case "ssh":
 198		candidates = append(candidates, "sshd.socket")
 199	case "sshd":
 200		candidates = append(candidates, "ssh.socket")
 201	}
 202
 203	seen := map[string]struct{}{}
 204	var sockets []string
 205	for _, candidate := range candidates {
 206		if _, ok := seen[candidate]; ok {
 207			continue
 208		}
 209		seen[candidate] = struct{}{}
 210		exist, _ := controller.CheckExist(candidate)
 211		if exist {
 212			sockets = append(sockets, candidate)
 213		}
 214	}
 215	return sockets

 315func handleSSHPortUpdate(oldValue, newValue string) {
 316	sudo := cmd.SudoHandleCmd()
 317	newPorts := splitSSHPorts(newValue)
 318	oldPorts := splitSSHPorts(oldValue)
 319
 320	stdout, _ := runWithOptionalSudo(sudo, "getenforce")
 321	if stdout == "Enforcing\n" {
 322		for _, port := range diffSSHPorts(oldPorts, newPorts) {
 323			if _, err := runWithOptionalSudo(sudo, "semanage", "port", "-d", "-t", "ssh_port_t", "-p", "tcp", port); err != nil {
 324				global.LOG.Warnf("remove selinux ssh port %s failed, err: %v", port, err)
 325			}
 326		}
 327		for _, port := range newPorts {
 328			_, _ = runWithOptionalSudo(sudo, "semanage", "port", "-a", "-t", "ssh_port_t", "-p", "tcp", port)
 329		}
 330	}
 331
 332	removedPorts, err := parseSSHPortsToInts(diffSSHPorts(oldPorts, newPorts))
 333	if err != nil {
 334		global.LOG.Errorf("parse removed ssh ports failed, err: %v", err)
 335	} else {
 336		addedPorts, err := parseSSHPortsToInts(diffSSHPorts(newPorts, oldPorts))
 337		if err != nil {
 338			global.LOG.Errorf("parse added ssh ports failed, err: %v", err)
 339		} else if err := OperateFirewallPort(removedPorts, addedPorts); err != nil {
 340			global.LOG.Errorf("reset firewall rules %s -> %s failed, err: %v", oldValue, newValue, err)
 341		}
 342	}
 343
 344	primaryPort, err := loadPrimarySSHPort(newValue)
 345	if err != nil {
 346		global.LOG.Errorf("load primary ssh port from %s failed, err: %v", newValue, err)
 347		return
 348	}
 349	if err := updateLocalConn(uint(primaryPort)); err != nil {
 350		global.LOG.Errorf("update local conn for terminal failed, err: %v", err)
 351	}
 352	if err := updateSSHSocketFile(strconv.Itoa(primaryPort)); err != nil {
 353		global.LOG.Errorf("update port for ssh.socket failed, err: %v", err)
 354	}
 355}

 368func diffSSHPorts(left, right []string) []string {
 369	rightSet := make(map[string]struct{}, len(right))
 370	for _, item := range right {
 371		rightSet[item] = struct{}{}
 372	}
 373	var diff []string
 374	for _, item := range left {
 375		if _, ok := rightSet[item]; ok {
 376			continue
 377		}
 378		diff = append(diff, item)
 379	}
 380	return diff
 381}

 412func (u *SSHService) SyncRootCert() error {
 413	currentUser, err := user.Current()
 414	if err != nil {
 415		return fmt.Errorf("load current user failed, err: %v", err)
 416	}
 417	sshDir := fmt.Sprintf("%s/.ssh", currentUser.HomeDir)
 418
 419	fileList, err := os.ReadDir(sshDir)
 420	if err != nil {
 421		return err
 422	}
 423	var rootCerts []model.RootCert
 424	fileMap := make(map[string]bool)
 425	for _, item := range fileList {
 426		if !item.IsDir() {
 427			fileMap[item.Name()] = true
 428		}
 429	}
 430	for item := range fileMap {
 431		if !strings.HasSuffix(item, ".pub") {
 432			continue
 433		}
 434		if !fileMap[strings.TrimSuffix(item, ".pub")] {
 435			continue
 436		}
 437		cert := model.RootCert{Name: strings.TrimSuffix(item, ".pub"), PublicKeyPath: path.Join(sshDir, item), PrivateKeyPath: path.Join(sshDir, strings.TrimSuffix(item, ".pub"))}
 438		pubItem, err := os.ReadFile(path.Join(sshDir, item))
 439		if err != nil {
 440			global.LOG.Errorf("read pubic key of %s for sync failed, err: %v", item, err)
 441			continue
 442		}
 443		cert.EncryptionMode = loadEncryptioMode(string(pubItem))
 444		rootCerts = append(rootCerts, cert)
 445	}
 446	return hostRepo.SyncCert(rootCerts)
 447}

 449func (u *SSHService) CreateRootCert(req dto.RootCertOperate) error {
 450	if cmd.CheckIllegal(req.EncryptionMode, req.PassPhrase) {
 451		return buserr.New("ErrCmdIllegal")
 452	}
 453	certItem, _ := hostRepo.GetCert(repo.WithByName(req.Name))
 454	if certItem.ID != 0 {
 455		return buserr.New("ErrRecordExist")
 456	}
 457	currentUser, err := user.Current()
 458	if err != nil {
 459		return fmt.Errorf("load current user failed, err: %v", err)
 460	}
 461	var cert model.RootCert
 462	if err := copier.Copy(&cert, req); err != nil {
 463		return err
 464	}
 465	privatePath := fmt.Sprintf("%s/.ssh/%s", currentUser.HomeDir, req.Name)
 466	publicPath := fmt.Sprintf("%s/.ssh/%s.pub", currentUser.HomeDir, req.Name)
 467	authFilePath := currentUser.HomeDir + "/.ssh/authorized_keys"
 468
 469	if info, err := os.Stat(authFilePath); err == nil && info.Size() > 0 {
 470		f, err := os.Open(authFilePath)
 471		if err != nil {
 472			return err
 473		}
 474		defer func() { _ = f.Close() }()
 475
 476		if _, err := f.Seek(-1, 2); err == nil {
 477			buf := make([]byte, 1)
 478			if _, err := f.Read(buf); err == nil && buf[0] != '\n' {
 479				appendFile, err := os.OpenFile(authFilePath, os.O_APPEND|os.O_WRONLY, 0600)
 480				if err != nil {
 481					return err
 482				}
 483				if _, err := appendFile.Write([]byte("\n")); err != nil {
 484					_ = appendFile.Close()
 485					return err
 486				}
 487				_ = appendFile.Close()
 488			}
 489		}
 490	}
 491
 492	if req.Mode == "input" || req.Mode == "import" {
 493		if err := os.WriteFile(privatePath, []byte(req.PrivateKey), constant.FilePerm); err != nil {
 494			return err
 495		}
 496		if err := os.WriteFile(publicPath, []byte(req.PublicKey), constant.FilePerm); err != nil {
 497			return err
 498		}
 499	} else {
 500		tmpPrivatePath := privatePath + ".tmp"
 501		tmpPublicPath := privatePath + ".tmp.pub"
 502		cmdMgr := cmd.NewCommandMgr(cmd.WithTimeout(2 * time.Minute))
 503		args := []string{
 504			"-t", req.EncryptionMode,
 505			"-f", tmpPrivatePath,
 506			"-N", req.PassPhrase,
 507			"-q",
 508		}
 509
 510		if err := cmdMgr.Run("ssh-keygen", args...); err != nil {
 511			_ = os.Remove(tmpPrivatePath)
 512			_ = os.Remove(tmpPublicPath)
 513			return fmt.Errorf("generate failed, %v", err)
 514		}
 515		if err := os.Rename(tmpPrivatePath, privatePath); err != nil {
 516			return fmt.Errorf("replace private key failed, %v", err)
 517		}
 518		if err := os.Rename(tmpPublicPath, publicPath); err != nil {
 519			return fmt.Errorf("replace public key failed, %v", err)
 520		}
 521	}
 522
 523	publicKeyBytes, err := os.ReadFile(publicPath)
 524	if err != nil {
 525		return fmt.Errorf("read public key failed, %v", err)
 526	}
 527
 528	cleanKey := strings.TrimRight(string(publicKeyBytes), "\n") + "\n"
 529	authFile, err := os.OpenFile(authFilePath, os.O_APPEND|os.O_WRONLY|os.O_CREATE, 0600)
 530	if err != nil {
 531		return fmt.Errorf("open authorized_keys failed, %v", err)
 532	}
 533	defer func() { _ = authFile.Close() }()
 534
 535	if _, err := authFile.Write([]byte(cleanKey)); err != nil {
 536		return fmt.Errorf("append authorized_keys failed, %v", err)
 537	}
 538
 539	cert.PrivateKeyPath = privatePath
 540	cert.PublicKeyPath = publicPath
 541	if len(cert.PassPhrase) != 0 {
 542		cert.PassPhrase, _ = encrypt.StringEncrypt(cert.PassPhrase)
 543	}
 544	return hostRepo.SaveCert(&cert)
 545}

 547func (u *SSHService) EditRootCert(req dto.RootCertOperate) error {
 548	currentUser, err := user.Current()
 549	if err != nil {
 550		return fmt.Errorf("load current user failed, err: %v", err)
 551	}
 552	certItem, _ := hostRepo.GetCert(repo.WithByID(req.ID))
 553	if certItem.ID == 0 {
 554		return buserr.New("ErrRecordNotFound")
 555	}
 556	oldPublicItem, err := os.ReadFile(certItem.PublicKeyPath)
 557	if err != nil {
 558		return err
 559	}
 560
 561	var cert model.RootCert
 562	if err := copier.Copy(&cert, req); err != nil {
 563		return err
 564	}
 565	cert.PrivateKeyPath = fmt.Sprintf("%s/.ssh/%s", currentUser.HomeDir, req.Name)
 566	cert.PublicKeyPath = fmt.Sprintf("%s/.ssh/%s.pub", currentUser.HomeDir, req.Name)
 567	if err := os.WriteFile(cert.PrivateKeyPath, []byte(req.PrivateKey), constant.FilePerm); err != nil {
 568		return err
 569	}
 570	if err := os.WriteFile(cert.PublicKeyPath, []byte(req.PublicKey), constant.FilePerm); err != nil {
 571		return err
 572	}
 573
 574	authFilePath := currentUser.HomeDir + "/.ssh/authorized_keys"
 575	authItem, err := os.ReadFile(authFilePath)
 576	if err != nil {
 577		return err
 578	}
 579	oldPublic := strings.ReplaceAll(string(oldPublicItem), "\n", "")
 580	newPublic := strings.ReplaceAll(string(req.PublicKey), "\n", "")
 581	lines := strings.Split(string(authItem), "\n")
 582	var newFiles []string
 583	for i := 0; i < len(lines); i++ {
 584		if len(lines[i]) != 0 && lines[i] != oldPublic && lines[i] != newPublic {
 585			newFiles = append(newFiles, lines[i])
 586		}
 587	}
 588	newFiles = append(newFiles, newPublic)
 589	if err := os.WriteFile(authFilePath, []byte(strings.Join(newFiles, "\n")), constant.FilePerm); err != nil {
 590		return fmt.Errorf("refresh authorized_keys failed, err: %v", err)
 591	}
 592	if len(cert.PassPhrase) != 0 {
 593		cert.PassPhrase, _ = encrypt.StringEncrypt(cert.PassPhrase)
 594	}
 595	return hostRepo.SaveCert(&cert)
 596}

 600	if err != nil {
 601		return 0, nil, err
 602	}
 603	var datas []dto.RootCert

 605		publicItem, err := os.ReadFile(records[i].PublicKeyPath)
 606		var publicBase64 string

 608			publicBase64 = base64.StdEncoding.EncodeToString(publicItem)
 609		}
 610		privateItem, _ := os.ReadFile(records[i].PrivateKeyPath)
 611		var privateBase64 string
 612		if err == nil && len(publicItem) != 0 {
 613			privateBase64 = base64.StdEncoding.EncodeToString(privateItem)
 614		}
 615		passPhrase, _ := encrypt.StringDecryptWithBase64(records[i].PassPhrase)
 616		datas = append(datas, dto.RootCert{
 617			ID:             records[i].ID,
 618			CreatedAt:      records[i].CreatedAt,
 619			Name:           records[i].Name,
 620			EncryptionMode: records[i].EncryptionMode,
 621			PassPhrase:     passPhrase,
 622			PublicKey:      publicBase64,
 623			PrivateKey:     privateBase64,
 624			Description:    records[i].Description,
 625		})
 626	}
 627	return total, datas, err
 628}
