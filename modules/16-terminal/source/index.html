 158func newHostSSHClient(host *model.Host, err error) (*ssh.SSHClient, error) {
 159	if err != nil {
 160		return nil, errors.WithMessage(err, "load host info by id failed")
 161	}
 162	connInfo := ssh.ConnInfo{
 163		Addr:       host.Addr,
 164		Port:       int(host.Port),
 165		User:       host.User,
 166		AuthMode:   host.AuthMode,
 167		Password:   host.Password,
 168		PrivateKey: []byte(host.PrivateKey),
 169	}
 170	if len(host.PassPhrase) != 0 {
 171		connInfo.PassPhrase = []byte(host.PassPhrase)
 172	}
 173	return ssh.NewClient(connInfo)

 259func loadDatabaseInitCmd(c *gin.Context) ([]string, error) {
 260	database := c.Query("database")
 261	databaseType := c.Query("databaseType")
 262	if len(databaseType) == 0 {
 263		return nil, fmt.Errorf("error param of database: %s or database type: %s", database, databaseType)
 264	}
 265	databaseConn, err := appInstallService.LoadConnInfo(dto.OperationWithNameAndType{Type: databaseType, Name: database})

 268	}
 269	if len(databaseConn.ContainerName) == 0 {
 270		return nil, fmt.Errorf("no such database container for database: %s or database type: %s", database, databaseType)
 271	}
 272	commands := []string{"exec", "-it", databaseConn.ContainerName}
 273	switch databaseType {
 274	case "mysql", "mysql-cluster":
 275		commands = append(commands, []string{"mysql", "-uroot", "-p" + databaseConn.Password}...)
 276	case "mariadb":
 277		commands = append(commands, []string{"mariadb", "-uroot", "-p" + databaseConn.Password}...)
 278	case "mongodb":
 279		commands = append(commands, []string{
 280			"mongosh",
 281			"--username", databaseConn.Username,
 282			"--password", databaseConn.Password,
 283			"--authenticationDatabase", "admin",
 284		}...)
 285	case "postgresql", "postgresql-cluster":
 286		commands = []string{"exec", "-e", fmt.Sprintf("PGPASSWORD=%s", databaseConn.Password), "-it", databaseConn.ContainerName, "psql", "-t", "-U", databaseConn.Username}
 287	}
 288
 289	return commands, nil

 292func wshandleError(ws *websocket.Conn, err error) bool {
 293	if err != nil {
 294		global.LOG.Errorf("handler ws faled:, err: %v", err)
 295		dt := time.Now().Add(time.Second)
 296		if ctlerr := ws.WriteControl(websocket.CloseMessage, []byte(err.Error()), dt); ctlerr != nil {
 297			wsData, err := json.Marshal(terminal.WsMsg{
 298				Type: terminal.WsMsgCmd,
 299				Data: base64.StdEncoding.EncodeToString([]byte(err.Error())),
 300			})
 301			if err != nil {
 302				_ = ws.WriteMessage(websocket.TextMessage, []byte("{\"type\":\"cmd\",\"data\":\"failed to encoding to json\"}"))
 303			} else {
 304				_ = ws.WriteMessage(websocket.TextMessage, wsData)
 305			}
 306		}
 307		return true
 308	}
 309	return false
 310}

  60func (ps *ProcessService) GetListeningProcess(c context.Context) ([]ListeningProcess, error) {

  65	"cmt">// One cache entry per (PID, socket type) so TCP and UDP sockets are not merged under one Protocol.
  66	type procKey struct {
  67		pid      int32
  68		protocol uint32
  69	}
  70	procCache := make(map[procKey]ListeningProcess, 64)
  71
  72	for _, conn := range conn {
  73		if conn.Pid == 0 {
  74			continue
  75		}
  76
  77		if (conn.Status == "LISTEN" && conn.Type == syscall.SOCK_STREAM) || (conn.Type == syscall.SOCK_DGRAM && conn.Raddr.Port == 0) {
  78			key := procKey{pid: conn.Pid, protocol: conn.Type}
  79			if _, exists := procCache[key]; !exists {
  80				proc, err := process.NewProcess(conn.Pid)
  81				if err != nil {
  82					continue
  83				}
  84				procData := ListeningProcess{
  85					PID: conn.Pid,
  86				}
  87				procData.Name, _ = proc.Name()
  88				procData.Port = make(map[uint32]struct{})
  89				procData.Port[conn.Laddr.Port] = struct{}{}
  90				procData.Protocol = conn.Type
  91				procCache[key] = procData
  92			} else {
  93				p := procCache[key]
  94				p.Port[conn.Laddr.Port] = struct{}{}
  95				procCache[key] = p
  96			}
  97		}
  98	}
  99
 100	procs := make([]ListeningProcess, 0, len(procCache))
 101	for _, proc := range procCache {
 102		procs = append(procs, proc)
 103	}
 104
 105	return procs, nil

 108func (ps *ProcessService) GetProcessInfoByPID(pid int32) (*websocket.PsProcessData, error) {
 109	p, err := process.NewProcess(pid)
 110	if err != nil {
 111		return nil, fmt.Errorf("get process info by pid %v: %v", pid, err)
 112	}
 113
 114	exists, err := p.IsRunning()
 115	if err != nil || !exists {
 116		return nil, fmt.Errorf("process %v is not running", pid)
 117	}
 118
 119	data := &websocket.PsProcessData{
 120		PID: pid,
 121	}
 122
 123	if name, err := p.Name(); err == nil {
 124		data.Name = name
 125	}
 126
 127	if ppid, err := p.Ppid(); err == nil {
 128		data.PPID = ppid
 129	}
 130
 131	if username, err := p.Username(); err == nil {
 132		data.Username = username
 133	}
 134
 135	if status, err := p.Status(); err == nil {
 136		if len(status) > 0 {
 137			data.Status = status[0]
 138		}
 139	}
 140
 141	if createTime, err := agentPsutil.NewProcessCreateTimeResolver().CreateTime(p); err == nil {
 142		data.StartTime = time.Unix(createTime/1000, 0).Format("2006-01-02 15:04:05")
 143	}
 144
 145	if numThreads, err := p.NumThreads(); err == nil {
 146		data.NumThreads = numThreads
 147	}
 148
 149	if connections, err := p.Connections(); err == nil {
 150		data.NumConnections = len(connections)
 151
 152		var connects []websocket.ProcessConnect

  54"cmt">//
  55"cmt">// On some LXC systems, gopsutil combines a container-relative /proc/uptime with
  56"cmt">// the host-relative starttime from /proc/<pid>/stat, which can place the start
  57"cmt">// time in the future. In that case, reading btime and starttime from /proc keeps
  58"cmt">// both values on the same clock base, matching the calculation used by ps.
  59func (r *ProcessCreateTimeResolver) CreateTime(proc *process.Process) (int64, error) {
  60	now := time.Now()
  61	createTime, createTimeErr := proc.CreateTime()
  62	if createTimeErr == nil && isValidProcessCreateTime(createTime, now) {
  63		return createTime, nil
  64	}
  65
  66	if r.bootTime > 0 && r.clockTicks > 0 {
  67		statPath := filepath.Join(r.procRoot, strconv.Itoa(int(proc.Pid)), "stat")
  68		if content, err := os.ReadFile(statPath); err == nil {
  69			if startTicks, err := parseProcessStartTicks(string(content)); err == nil {
  70				startMillis := startTicks * 1000 / r.clockTicks
  71				fallbackTime := r.bootTime*1000 + int64(startMillis)
  72				if isValidProcessCreateTime(fallbackTime, now) {
  73					return fallbackTime, nil
  74				}
  75			}
  76		}
  77	}
  78	if createTimeErr != nil {
  79		return 0, fmt.Errorf("resolve process create time: %w", createTimeErr)
  80	}
  81	return 0, fmt.Errorf("invalid process create time: %d", createTime)
  82}

 116func parseProcessStartTicks(stat string) (uint64, error) {
 117	"cmt">// The second field (comm) is wrapped in parentheses and may contain spaces
 118	"cmt">// or parentheses, so split only after its closing parenthesis. The remaining
 119	"cmt">// fields begin at field 3 (state), making starttime (field 22) index 19.
