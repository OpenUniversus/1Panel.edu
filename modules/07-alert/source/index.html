 366func (a *AlertRepo) SyncAll(data []model.AlertConfig) error {
 367	tx := global.AlertDB.Begin()
 368	if tx.Error != nil {
 369		return tx.Error
 370	}
 371	defer func() {
 372		if r := recover(); r != nil {
 373			tx.Rollback()
 374			panic(r)
 375		}
 376	}()
 377
 378	var oldConfigs []model.AlertConfig
 379	if err := tx.Find(&oldConfigs).Error; err != nil {
 380		tx.Rollback()
 381		return err
 382	}
 383
 384	usedConfigIDs, err := loadUsedAlertConfigIDs(tx)
 385	if err != nil {
 386		tx.Rollback()
 387		return err
 388	}
 389
 390	oldConfigMap := make(map[string]uint)
 391	oldConfigByType := make(map[string][]model.AlertConfig)
 392	oldConfigByKey := make(map[string][]model.AlertConfig)
 393	consumedConfigIDs := make(map[uint]struct{})
 394	for _, item := range oldConfigs {
 395		if singletonTypes[item.Type] {
 396			oldConfigMap[item.Type] = item.ID
 397			continue
 398		}
 399		oldConfigByType[item.Type] = append(oldConfigByType[item.Type], item)
 400		oldConfigByKey[alertConfigSyncKey(item)] = append(oldConfigByKey[alertConfigSyncKey(item)], item)
 401	}
 402	for _, item := range data {
 403		if singletonTypes[item.Type] {
 404			if val, ok := oldConfigMap[item.Type]; ok {
 405				item.ID = val
 406				delete(oldConfigMap, item.Type)
 407				consumedConfigIDs[item.ID] = struct{}{}
 408			} else {
 409				item.ID = 0
 410			}
 411			if item.ID == 0 {
 412				if err := tx.Create(&item).Error; err != nil {
 413					tx.Rollback()
 414					return err
 415				}
 416			} else if err := tx.Save(&item).Error; err != nil {
 417				tx.Rollback()
 418				return err
 419			}
 420			continue
 421		}
 422
 423		key := alertConfigSyncKey(item)
 424		if matched, ok := popAlertConfigByKey(oldConfigByKey, key); ok {
 425			item.ID = matched.ID
 426			consumedConfigIDs[item.ID] = struct{}{}
 427			if err := tx.Save(&item).Error; err != nil {
 428				tx.Rollback()
 429				return err
 430			}
 431			deleteAlertConfigByID(oldConfigByType, matched.ID)
 432			continue
 433		}
 434
 435		if matched, ok := popUnusedAlertConfigByType(oldConfigByType, usedConfigIDs, item.Type); ok {
 436			item.ID = matched.ID
 437			consumedConfigIDs[item.ID] = struct{}{}
 438			if err := tx.Save(&item).Error; err != nil {
 439				tx.Rollback()
 440				return err
 441			}
 442			continue
 443		}
 444
 445		item.ID = 0
 446		if err := tx.Create(&item).Error; err != nil {
 447			tx.Rollback()
 448			return err
 449		}
 450	}
 451	for _, item := range oldConfigs {
 452		if _, used := usedConfigIDs[item.ID]; used {
 453			continue
 454		}
 455		if _, kept := consumedConfigIDs[item.ID]; kept {
 456			continue
 457		}
 458		if err := tx.Where("id = ?", item.ID).Delete(&model.AlertConfig{}).Error; err != nil {
 459			tx.Rollback()
 460			return err
 461		}
 462	}
 463	if err := tx.Commit().Error; err != nil {
 464		tx.Rollback()
 465		return err
 466	}
 467	return nil
 468}

 268func (a AlertService) GetDisks() ([]dto.DiskDTO, error) {
 269	var disks []dto.DiskDTO
 270	excludes := map[string]struct{}{
 271		"/mnt/cdrom": {}, "/boot": {}, "/boot/efi": {}, "/dev": {}, "/dev/shm": {},
 272		"/run/lock": {}, "/run": {}, "/run/shm": {}, "/run/user": {},
 273	}
 274	stdout, err := executeDiskCommand()
 275	if err != nil {
 276		return disks, nil
 277	}
 278
 279	lines := strings.Split(stdout, "\n")
 280	var mounts []dto.AlertDiskInfo
 281
 282	for _, line := range lines {
 283		fields := strings.Fields(line)
 284		if len(fields) < 7 {
 285			continue
 286		}
 287		mountPoint := strings.Join(fields[6:], " ")
 288		if shouldExclude(fields, mountPoint, excludes) {
 289			continue
 290		}
 291		mounts = append(mounts, dto.AlertDiskInfo{Type: fields[1], Device: fields[0], Mount: mountPoint})
 292
 293	}
 294
 295	var (
 296		wg sync.WaitGroup
 297		mu sync.Mutex
 298	)
 299	wg.Add(len(mounts))
 300	for i := 0; i < len(mounts); i++ {
 301		go func(timeoutCh <-chan time.Time, mount dto.AlertDiskInfo) {
 302			defer wg.Done()
 303
 304			var itemData dto.DiskDTO
 305			itemData.Path = mount.Mount
 306			itemData.Type = mount.Type
 307			itemData.Device = mount.Device
 308			select {
 309			case <-timeoutCh:
 310				mu.Lock()
 311				disks = append(disks, itemData)
 312				mu.Unlock()
 313				global.LOG.Errorf("load disk info from %s failed, err: timeout", mount.Mount)
 314			default:
 315				state, err := disk.Usage(mount.Mount)
 316				if err != nil {
 317					mu.Lock()
 318					disks = append(disks, itemData)
 319					mu.Unlock()
 320					global.LOG.Errorf("load disk info from %s failed, err: %v", mount.Mount, err)
 321					return
 322				}
 323				itemData.Total = state.Total
 324				itemData.Free = state.Free
 325				itemData.Used = state.Used
 326				itemData.UsedPercent = state.UsedPercent
 327				itemData.InodesTotal = state.InodesTotal
 328				itemData.InodesUsed = state.InodesUsed
 329				itemData.InodesFree = state.InodesFree
 330				itemData.InodesUsedPercent = state.InodesUsedPercent
 331				mu.Lock()
 332				disks = append(disks, itemData)
 333				mu.Unlock()
 334			}
 335		}(time.After(5*time.Second), mounts[i])
 336	}
 337	wg.Wait()
 338
 339	sort.Slice(disks, func(i, j int) bool {
 340		return disks[i].Path < disks[j].Path
 341	})
 342	return disks, nil
 343}

 345func executeDiskCommand() (string, error) {
 346	cmdMgr := cmd.NewCommandMgr(cmd.WithTimeout(2 * time.Second))
 347	stdout, err := cmdMgr.RunWithStdout("df", "-hT", "-P")
 348	if err != nil {
 349		cmdMgr2 := cmd.NewCommandMgr(cmd.WithTimeout(1 * time.Second))
 350		stdout, err = cmdMgr2.RunWithStdout("df", "-lhT", "-P")
 351	}
 352	if err != nil {
 353		return stdout, err
 354	}
 355	var lines []string
 356	for _, line := range strings.Split(stdout, "\n") {
 357		if !strings.Contains(line, "/") || strings.Contains(line, "tmpfs") || strings.Contains(line, "snap/core") || strings.Contains(line, "udev") {
 358			continue
 359		}
 360		lines = append(lines, line)
 361	}
 362	if len(lines) == 0 {
 363		return "", nil
 364	}
 365	return strings.Join(lines, "\n"), nil
 366}

 590func isIPInWhitelist(ip string, whitelist []string) bool {
 591	targetIP := net.ParseIP(strings.TrimSpace(ip))
 592	if targetIP == nil {
 593		return false
 594	}
 595	for _, item := range whitelist {
 596		item = strings.TrimSpace(item)
 597		if item == "" {
 598			continue
 599		}
 600		if item == ip {
 601			return true
 602		}
 603		if whiteIP := net.ParseIP(item); whiteIP != nil {
 604			if whiteIP.Equal(targetIP) {
 605				return true
 606			}
 607			continue
 608		}
 609		_, ipNet, err := net.ParseCIDR(item)
 610		if err != nil {
 611			continue
 612		}
 613		if ipNet.Contains(targetIP) {
 614			return true
 615		}
 616	}
 617	return false
 618}
