 600func getEnvMapByPath(envPath string) (map[string]interface{}, error) {
 601	envMap := make(map[string]interface{})
 602	envFile, err := os.ReadFile(envPath)
 603	if err != nil {
 604		return envMap, err
 605	}
 606	lines := strings.Split(string(envFile), "\n")
 607	for _, line := range lines {
 608		line = strings.TrimSpace(line)
 609		if line == "" || strings.HasPrefix(line, "#") {
 610			continue
 611		}
 612		parts := strings.SplitN(line, "=", 2)
 613		if len(parts) == 2 {
 614			value := strings.TrimSpace(parts[1])
 615			if len(value) >= 2 {
 616				if (value[0] == '"' && value[len(value)-1] == '"') || (value[0] == '\'' && value[len(value)-1] == '\'') {
 617					value = value[1 : len(value)-1]
 618				}
 619			}
 620			envMap[parts[0]] = value
 621		}
 622	}
 623	return envMap, nil
 624}

 146	backupTask.AddSubTaskWithOps(i18n.GetMsgByKey("ComposeBackupContainers"), func(t *task.Task) error { return nil }, nil, 3, time.Hour)
 147	for _, containerName := range containerNames {
 148		backupFileName := fmt.Sprintf("%s.tar.gz", sanitizeComposeFileName(containerName))
 149		backupFile := path.Join(composeCtx.tmpDir, "containers", backupFileName)
 150		if err := handleContainerBackup(containerName, backupTask, 0, path.Dir(backupFile), path.Base(backupFile), "", "", false); err != nil {
 151			return err
 152		}
 153		composeCtx.meta.Containers = append(composeCtx.meta.Containers, path.Join("containers", backupFileName))
 154	}
 155	backupTask.AddSubTaskWithOps(i18n.GetMsgByKey("ComposeBackupMeta"), func(t *task.Task) error { return stepWriteComposeBackupMeta(composeCtx) }, nil, 3, time.Hour)
 156	backupTask.AddSubTaskWithOps(task.GetTaskName(composeCtx.composeName, task.TaskBackup, task.TaskScopeBackup), func(t *task.Task) error {
 157		return stepPackComposeBackup(composeCtx)
 158	}, nil, 3, time.Hour)
 159	if req.StopBefore {
 160		backupTask.AddSubTaskWithOps(i18n.GetMsgByKey("ComposeBackupStart"), func(t *task.Task) error {
 161			return stepStartComposeAfterBackup(composeCtx)
 162		}, nil, 3, time.Hour)
 163	}
 164	backupTask.AddSubTaskWithOps(i18n.GetMsgByKey("ComposeBackupCleanup"), func(t *task.Task) error {
 165		composeCtx.close()
 166		return nil
 167	}, nil, 0, time.Hour)
 168	if parentTask != nil {
 169		return nil
 170	}
 171	go func() {
 172		defer composeCtx.close()
 173		if err := backupTask.Execute(); err != nil {
 174			markBackupFailed(recordID, err)
 175			return
 176		}
 177		backupRepo.UpdateRecordByMap(recordID, map[string]interface{}{"status": constant.StatusSuccess})
 178	}()
 179	return nil
 180}

 301			}
 302			return nil
 303		}, nil, 0, timeout)

 305		return nil
 306	}, nil, 3, timeout)

 309	}

 312	}()
 313	return nil

 354func loadComposePathAndFiles(composeName string, dockerClient *client.Client) (string, []string, error) {
 355	composeRecord, _ := composeRepo.GetRecord(repo.WithByName(composeName))
 356	if composeRecord.ID == 0 {
 357		composeRecord, _ = composeRepo.GetRecord(repo.WithByName(strings.ToLower(composeName)))
 358	}
 359	composePath := composeRecord.Path
 360	if composePath == "" {
 361		options := container.ListOptions{All: true}
 362		options.Filters = filters.NewArgs(filters.Arg("label", composeProjectLabel))
 363		list, err := dockerClient.ContainerList(context.Background(), options)
 364		if err != nil {
 365			return "", nil, err
 366		}
 367		if len(list) == 0 {
 368			return "", nil, fmt.Errorf("compose %s not found", composeName)
 369		}
 370		var targetContainer *container.Summary
 371		for i := range list {
 372			if strings.EqualFold(list[i].Labels[composeProjectLabel], composeName) {
 373				targetContainer = &list[i]
 374				break
 375			}
 376		}
 377		if targetContainer == nil {
 378			return "", nil, fmt.Errorf("compose %s not found", composeName)
 379		}
 380		config := targetContainer.Labels[composeConfigLabel]
 381		workdir := targetContainer.Labels[composeWorkdirLabel]
 382		if len(config) != 0 && len(workdir) != 0 && strings.Contains(config, workdir) {
 383			composePath = config
 384		} else {
 385			composePath = workdir
 386		}
 387	}
 388	composeFiles := normalizeComposeFiles(composePath)
 389	if len(composeFiles) == 0 {
 390		return "", nil, fmt.Errorf("compose file not found for %s", composeName)
 391	}
 392	return composePath, composeFiles, nil
 393}

 395func normalizeComposeFiles(composePath string) []string {
 396	items := strings.Split(composePath, ",")
 397	result := make([]string, 0)
 398	seen := make(map[string]struct{})
 399	for _, item := range items {
 400		item = strings.TrimSpace(item)
 401		if item == "" {
 402			continue
 403		}
 404		stat, err := os.Stat(item)
 405		if err == nil && stat.IsDir() {
 406			item = path.Join(item, "docker-compose.yml")
 407		}
 408		if _, err := os.Stat(item); err != nil {
 409			continue
 410		}
 411		if _, ok := seen[item]; ok {
 412			continue
 413		}
 414		seen[item] = struct{}{}
 415		result = append(result, item)
 416	}
 417	sort.Strings(result)
 418	return result
 419}

 497		}

 499			OriginalPath: filePath,
 500			FileName:     path.Base(filePath),
 501			RelativePath: relativePath,
 502			BackupPath:   path.Join("compose_files", backupName),
 503		})
 504	}
 505	if len(composeCtx.composeFiles) != 0 {
 506		envPath := path.Join(path.Dir(composeCtx.composeFiles[0]), ".env")
 507		if composeCtx.fileOp.Stat(envPath) {

 513				return err
 514			}
 515		}
 516	}
 517	return nil
 518}

 606func safeJoinWithinBase(baseDir, name string) (string, error) {
 607	base := filepath.Clean(baseDir)
 608	candidate := strings.TrimSpace(name)
 609	candidate = strings.ReplaceAll(candidate, "\\", "/")
 610	candidate = filepath.Clean(filepath.FromSlash(candidate))
 611	if candidate == "" || candidate == "." {
 612		return "", fmt.Errorf("invalid path: empty")
 613	}
 614	if filepath.IsAbs(candidate) {
 615		return "", fmt.Errorf("invalid path %q: absolute path is not allowed", name)
 616	}
 617	if candidate == ".." || strings.HasPrefix(candidate, ".."+string(filepath.Separator)) {
 618		return "", fmt.Errorf("invalid path %q: path escapes base directory", name)
 619	}
 620	resolved := filepath.Clean(filepath.Join(base, candidate))
 621	rel, err := filepath.Rel(base, resolved)
 622	if err != nil {
 623		return "", fmt.Errorf("resolve path %q failed, err: %v", name, err)
 624	}
 625	if rel == ".." || strings.HasPrefix(rel, ".."+string(filepath.Separator)) {
 626		return "", fmt.Errorf("invalid path %q: path escapes base directory", name)
 627	}
 628	return resolved, nil
 629}

 631func stepRestoreComposeFiles(recoverCtx *composeRecoverContext) error {
 632	if recoverCtx.targetDir != "" {
 633		_ = os.RemoveAll(recoverCtx.targetDir)
 634	}
 635	if err := os.MkdirAll(recoverCtx.targetDir, os.ModePerm); err != nil {
 636		return err
 637	}
 638	restored := make([]string, 0, len(recoverCtx.meta.Files))
 639	for _, item := range recoverCtx.meta.Files {
 640		backupPath, err := safeJoinWithinBase(recoverCtx.tmpDir, item.BackupPath)
 641		if err != nil {
 642			return fmt.Errorf("invalid compose backup path %q, err: %v", item.BackupPath, err)
 643		}
 644		if !recoverCtx.fileOp.Stat(backupPath) {
 645			continue
 646		}
 647		targetName := item.FileName
 648		if item.RelativePath != "" {
 649			targetName = item.RelativePath
 650		}
 651		if targetName == "" {
 652			targetName = path.Base(item.OriginalPath)
 653		}
 654		if targetName == "" {
 655			targetName = "docker-compose.yml"
 656		}
 657		targetPath, err := safeJoinWithinBase(recoverCtx.targetDir, targetName)
 658		if err != nil {
 659			return fmt.Errorf("invalid compose target path %q, err: %v", targetName, err)
 660		}
 661		if err := os.MkdirAll(path.Dir(targetPath), os.ModePerm); err != nil {
 662			return err
 663		}
 664		content, err := os.ReadFile(backupPath)
 665		if err != nil {
 666			return err
 667		}
 668		if err := recoverCtx.fileOp.SaveFile(targetPath, string(content), fs.ModePerm); err != nil {
 669			return err
 670		}
 671		restored = append(restored, targetPath)
 672	}
 673	envPath := path.Join(recoverCtx.tmpDir, "compose_files", ".env")
 674	if recoverCtx.fileOp.Stat(envPath) {
 675		envContent, err := os.ReadFile(envPath)
 676		if err != nil {
 677			return err
 678		}
 679		if err := recoverCtx.fileOp.SaveFile(path.Join(recoverCtx.targetDir, ".env"), string(envContent), fs.ModePerm); err != nil {
 680			return err
 681		}
 682	}
 683	if len(restored) == 0 {
 684		defaultPath := path.Join(recoverCtx.targetDir, "docker-compose.yml")
 685		if !recoverCtx.fileOp.Stat(defaultPath) {
 686			return fmt.Errorf("compose file not found in backup data")
 687		}
 688		restored = append(restored, defaultPath)
 689	}
 690	sort.Strings(restored)
 691	recoverCtx.composePath = strings.Join(restored, ",")
 692	return nil
 693}

 422			Type:        string(item.Type),

 425			Destination: item.Destination,

 457			} else {

 459					return err
 460				}
 461			}

 316			_ = os.RemoveAll(dstDir)
 317			return "", err
 318		}
 319	}
 320	global.LOG.Infof("decompress file %s successful, now start to check test.sql is exist", file)
 321	var sqlFiles []string

 323	_ = filepath.Walk(dstDir, func(path string, info os.FileInfo, err error) error {
 324		if err != nil {
 325			return nil
 326		}
 327		if !info.IsDir() && strings.HasSuffix(info.Name(), ".sql") {

 329			if info.Name() == "test.sql" {
 330				hasTestSql = true
 331			}
 332		}
 333		return nil
 334	})
 335	if len(sqlFiles) == 1 {
 336		return sqlFiles[0], nil
 337	}
 338	if !hasTestSql {
 339		_ = os.RemoveAll(dstDir)
 340		return "", fmt.Errorf("no such file named test.sql in %s", fileName)
